## 🧠 L'Obiettivo Didattico: La Funzione di Reward

Ad ogni step ($0{,}02\text{ secondi}$), l'agente riceve gli 8 sensori fisici di bordo (`obs`), il comando motore (`action`) e gli eventi di fine volo:

```python
def custom_reward(obs, action, is_success, is_crashed, is_timeout):
    x, y, vx, vy, angle, vangle, left_leg, right_leg = obs
    reward = 0.0

    # =========================================================================
    # PROGETTA QUI LA TUA FORMULA DI REWARD SHAPING
    # =========================================================================

    # Esempi di termini da considerare:
    # - Posizione orizzontale: avvicinarsi al centro (x = 0.0)
    # - Velocità di discesa: frenare quando vy è troppo negativo
    # - Assetto angolare: mantenere il modulo verticale (angle = 0.0)
    # - Efficienza: premiare lo spegnimento dei motori a terra (action == 0)

    # Eventi terminali
    if is_success:
        reward += 100.0
    if is_crashed:
        reward -= 100.0

    return float(reward)
```

### Le 2 Trappole Matematiche Classiche
Gli studenti imparano ad analizzare il fenomeno del *Reward Hacking*:
1. 💥 **La Trappola Kamikaze (Suicidio Rapido):** Se le penalità per step in volo sono troppo elevate, all'agente conviene schiantarsi subito per interrompere l'episodio ed evitare di accumulare penalità continue.
2. 🪂 **La Trappola dell'Hovering Infinito:** Se l'agente riceve premi positivi in volo o non viene penalizzato il timeout, sceglie di galleggiare a mezz'aria senza mai rischiare l'atterraggio.

## 📊 Telemetria, Diagnostica e Reportistica

Al termine di ogni sessione di addestramento:

def _build_reward_caller(reward_fn: Callable) -> Callable:
    """
    Ispeziona una sola volta la firma della funzione dello studente e restituisce
    un dispatcher specializzato ad alta velocità (zero controlli in step()).
    """
    try:
        param_count = len(inspect.signature(reward_fn).parameters)
    except Exception:
        param_count = 5

    if param_count == 5:
        return lambda obs, act, succ, crash, tout: reward_fn(obs, act, succ, crash, tout)
    if param_count == 4:
        return lambda obs, act, succ, crash, tout: reward_fn(obs, succ, crash, tout)
    if param_count == 2:
        return lambda obs, act, succ, crash, tout: reward_fn(obs, act)
    if param_count == 1:
        return lambda obs, act, succ, crash, tout: reward_fn(obs)
    if param_count == 12:
        return lambda obs, act, succ, crash, tout: reward_fn(*obs, act, succ, crash, tout)
    if param_count == 11:
        return lambda obs, act, succ, crash, tout: reward_fn(*obs, succ, crash, tout)

    def _fallback(obs, act, succ, crash, tout):
        try:
            return reward_fn(obs, act, succ, crash, tout)
        except TypeError:
            return reward_fn(obs, succ, crash, tout)

    return _fallback


class CustomRewardWrapper(gym.Wrapper if _GYM_AVAILABLE else object):
    """
    Wrapper ad alte prestazioni che intercetta ogni `step()`, calcola la telemetria
    di volo e delega il calcolo della ricompensa alla funzione dello studente.
    Popola il dizionario `info` solo a fine episodio per minimizzare allocazioni e IPC.
    """

    def __init__(self, env: gym.Env, reward_fn: Callable):
        super().__init__(env)
        self.reward_fn = reward_fn
        self._call_reward = _build_reward_caller(reward_fn)
        self._unwrapped = env.unwrapped

        self.ep_fuel_used: float = 0.0
        self.ep_main_engine_steps: int = 0
        self.ep_side_engine_steps: int = 0
        self.ep_air_steps: int = 0
        self.ep_ground_steps: int = 0

    def reset(self, **kwargs):
        self.ep_fuel_used = 0.0
        self.ep_main_engine_steps = 0
        self.ep_side_engine_steps = 0
        self.ep_air_steps = 0
        self.ep_ground_steps = 0
        return self.env.reset(**kwargs)

    def step(self, action):
        obs, original_reward, terminated, truncated, info = self.env.step(action)
        act_int = int(action)

        # 1. Telemetria leggera in-place (zero allocazioni)
        if act_int == 2:
            self.ep_fuel_used += FUEL_MAIN_ENGINE
            self.ep_main_engine_steps += 1
        elif act_int == 1 or act_int == 3:
            self.ep_fuel_used += FUEL_SIDE_ENGINE
            self.ep_side_engine_steps += 1

        # Accesso scalare diretto (.item) evita allocazione di np.float32
        if obs.item(6) > 0.5 or obs.item(7) > 0.5:
            self.ep_ground_steps += 1
        else:
            self.ep_air_steps += 1

        # 2. Fast-path per oltre il 99.5% degli step (non terminali)
        if not (terminated or truncated):
            try:
                custom_rew = float(self._call_reward(obs, act_int, False, False, False))
            except Exception as e:
                raise RuntimeError(
                    f"[Play2Learn Error] Errore nell'esecuzione della tua 'custom_reward': {e}\n"
                    f"Controlla i calcoli matematici (es. divisioni per zero o indici errati)."
                ) from e
            return obs, custom_rew, False, False, info

        # 3. Stato terminale / timeout (eseguito 1 volta per episodio)
        is_timeout = bool(truncated)
        is_crashed = False
        is_success = False
        is_off_target = False

        if terminated:
            unwrapped = self._unwrapped
            game_over = getattr(unwrapped, "game_over", False)
            lander = getattr(unwrapped, "lander", None)
            is_awake = getattr(lander, "awake", True) if lander is not None else True

            final_x = abs(obs.item(0))
            final_angle = abs(obs.item(4))
            legs_contact = (obs.item(6) > 0.5) or (obs.item(7) > 0.5)

            # 1. Schianto catastrofico evidente: scafo tocca terra (game_over) o fuori schermo (|x| >= 1.0)
            if game_over or final_x >= BOUNDARY_X:
                is_crashed = True
                is_success = False
            # 2. Il modulo ha toccato terra e si è fermato (Box2D sleep / awake=False) o fine episodio positiva Gym
            elif (not is_awake) or (original_reward >= 100.0):
                # Per essere un vero SUCCESSO didattico e fisico, l'atterraggio DEVE avvenire:
                # - Sulla piazzola delimitata dalle bandierine (|x| <= 0.20)
                # - Con assetto verticale stabile (|angle| <= 0.25 rad, ~14°)
                # - Con contatto delle zampe
                on_helipad = final_x <= HELIPAD_X_MAX
                upright = final_angle <= SAFE_LANDING_ANGLE

                if on_helipad and upright and legs_contact:
                    is_success = True
                    is_crashed = False
                else:
                    # Atterraggio completato ma fuori piazzola (es. su crateri/pendii) o ribaltato:
                    # è un fallimento/fuori bersaglio!
                    is_success = False
                    is_crashed = True
                    if not on_helipad:
                        is_off_target = True
            else:
                # Caso residuale
                on_helipad = final_x <= HELIPAD_X_MAX
                upright = final_angle <= SAFE_LANDING_ANGLE
                if on_helipad and upright and legs_contact and original_reward > 0.0:
                    is_success = True
                    is_crashed = False
                else:
                    is_crashed = True
                    is_success = False
                    if not on_helipad:
                        is_off_target = True

        try:
            custom_rew = float(
                self._call_reward(obs, act_int, is_success, is_crashed, is_timeout)
            )
        except Exception as e:
            raise RuntimeError(
                f"[Play2Learn Error] Errore nell'esecuzione della tua 'custom_reward': {e}\n"
                f"Controlla i calcoli matematici (es. divisioni per zero o indici errati)."
            ) from e

        # Popolamento `info` concentrato a fine episodio
        info["custom_reward"] = custom_rew
        info["original_gym_reward"] = float(original_reward)
        info["is_success"] = is_success
        info["is_crashed"] = is_crashed
        info["is_timeout"] = is_timeout
        info["is_off_target"] = is_off_target
        info["telemetry"] = {
            "fuel_used": float(self.ep_fuel_used),
            "main_engine_steps": self.ep_main_engine_steps,
            "side_engine_steps": self.ep_side_engine_steps,
            "air_steps": self.ep_air_steps,
            "ground_steps": self.ep_ground_steps,
            "final_abs_x": abs(obs.item(0)),
            "final_abs_y": abs(obs.item(1)),
            "final_abs_vx": abs(obs.item(2)),
            "final_abs_vy": abs(obs.item(3)),
            "final_abs_angle": abs(obs.item(4)),
        }

        return obs, custom_rew, terminated, truncated, info


def _make_single_env(
    env_id: str,
    reward_fn: Callable,
    seed: int,
    rank: int,
    env_kwargs: dict[str, Any],
):
    """Costruisce un singolo ambiente isolato, senza PassiveEnvChecker."""
    def _init() -> gym.Env:
        try:
            env = gym.make(env_id, disable_env_checker=True, **env_kwargs)
        except TypeError:
            env = gym.make(env_id, **env_kwargs)
        env = CustomRewardWrapper(env, reward_fn)
        env = Monitor(env, info_keywords=("is_success",))
        env.reset(seed=seed + rank)
        return env
    return _init


def get_optimal_vec_config(
    n_envs: int | None = None,
    total_timesteps: int = 200_000,
    prefer_subproc: bool | None = None,
) -> tuple[int, bool]:
    """
    Determina la combinazione ottimale (n_envs, use_subproc) per l'hardware corrente:
    - Colab / CPU <= 4 thread: DummyVecEnv con 4 env (massimo FPS, zero IPC).
    - Windows (spawn lento e fragile in Jupyter): DummyVecEnv con 4 env di default.
    - Linux multi-core (>= 6 thread) su run >= 250k step: SubprocVecEnv con 8 env.
    """
    cpu_count = os.cpu_count() or 2
    is_windows = sys.platform.startswith("win")

    if prefer_subproc is not None:
        use_subproc = prefer_subproc
    else:
        use_subproc = (not is_windows) and (cpu_count >= 6) and (total_timesteps >= 250_000)

    if n_envs is None:
        resolved_envs = 8 if use_subproc else 4
    else:
        resolved_envs = max(1, int(n_envs))

    if resolved_envs == 1:
        use_subproc = False

    return resolved_envs, use_subproc


def make_parallel_envs(
    env_id: str,
    reward_fn: Callable,
    n_envs: int | None = None,
    seed: int = 42,
    env_kwargs: dict[str, Any] | None = None,
    prefer_subproc: bool | None = None,
    total_timesteps: int = 200_000,
) -> VecEnv:
    """
    Crea la VecEnv parallela ottimizzata per Google Colab (2 vCPU) e desktop locali.
    """
    env_kwargs = env_kwargs or {}
    resolved_envs, use_subproc = get_optimal_vec_config(
        n_envs=n_envs,
        total_timesteps=total_timesteps,
        prefer_subproc=prefer_subproc,
    )
    env_fns = [
        _make_single_env(env_id, reward_fn, seed, i, env_kwargs)
        for i in range(resolved_envs)
    ]

    if use_subproc:
        try:
            start_method = "forkserver" if not sys.platform.startswith("win") else "spawn"
            return SubprocVecEnv(env_fns, start_method=start_method)
        except Exception:
            return DummyVecEnv(env_fns)

    return DummyVecEnv(env_fns)


def wrap_vec_normalize(
    vec_env: VecEnv,
    gamma: float | None = None,
) -> VecEnv:
    """
    Applica VecNormalize SOLO sulle ricompense (norm_obs=False, norm_reward=True).
    Stabilizza PPO con qualsiasi scala di custom_reward senza alterare le osservazioni
    e mantenendo intatta la compatibilità a inferenza senza file di statistiche extra.
    """
    params = dict(VECNORM_PARAMS)
    if gamma is not None:
        params["gamma"] = float(gamma)
    return VecNormalize(vec_env, **params)

"""

telemetria O(1), best-model checkpointing in RAM e progress bar leggera per Colab/Jupyter.

    from stable_baselines3 import PPO
    from stable_baselines3.common.callbacks import BaseCallback
    from stable_baselines3.common.vec_env import VecEnv

class LinearSchedule:
    """
    Schedule lineare top-level pienamente serializzabile da pickle/cloudpickle
    per PPO.save() e PPO.load() senza warning.
    """

    def __init__(self, initial_value: float, final_value: float):
        self.initial_value = float(initial_value)
        self.final_value = float(final_value)

    def __call__(self, progress_remaining: float) -> float:
        return self.final_value + progress_remaining * (self.initial_value - self.final_value)


class RewardTrackerCallback(BaseCallback):
    """
    Callback ad alte prestazioni:
    - Traccia telemetria completa per Diagnostics e Visualizer.
    - Calcola la media mobile del successo in O(1) con deque.
    - Salva i pesi migliori SOLO in RAM (~60 KB) durante il training, azzerando
      l'I/O sincrono su disco; scrive best_model.zip una sola volta a fine run.
    - Mostra una progress bar HTML/console throttled (ogni 1.5s) senza lag su Colab.
    """

    def __init__(
        self,
        best_model_save_dir: Path | str | None = None,
        total_timesteps: int = 0,
        show_progress: bool = True,
        progress_interval_sec: float = 1.5,
        window_size: int = 30,
        capture_milestones: bool = True,
        verbose: int = 0,
    ):
        super().__init__(verbose)
        self.best_model_save_dir = Path(best_model_save_dir) if best_model_save_dir else None
        self.total_timesteps = int(total_timesteps)
        self.show_progress = show_progress
        self.progress_interval_sec = float(progress_interval_sec)
        self.window_size = int(window_size)
        self.capture_milestones = capture_milestones

        # Snapshot intermedi per timelapse di evoluzione apprendimento (RAM, overhead 0)
        self.milestone_snapshots: list[dict[str, Any]] = []
        self._milestone_targets: list[float] = [0.10, 0.35, 0.65]
        self._next_milestone_idx: int = 0

        # Liste telemetriche pubbliche (contratto con diagnostics.py e visualizer.py)
        self.episode_rewards: list[float] = []
        self.episode_lengths: list[int] = []
        self.outcomes: list[str] = []
        self.fuel_used_list: list[float] = []
        self.air_steps_list: list[int] = []
        self.ground_steps_list: list[int] = []
        self.final_x_list: list[float] = []
        self.final_vy_list: list[float] = []
        self.final_angle_list: list[float] = []

        self.successes: int = 0
        self.crashes: int = 0
        self.timeouts: int = 0

        # Finestra scorrevole O(1)
        self._window_outcomes: deque[int] = deque(maxlen=self.window_size)
        self._window_fuels: deque[float] = deque(maxlen=self.window_size)
        self._window_successes: int = 0

        # Snapshot in RAM del miglior modello
        self.best_moving_success_rate: float = -1.0
        self._best_moving_fuel: float = float("inf")
        self._best_policy_state: dict[str, torch.Tensor] | None = None
        self.best_model_path: Path | None = None

        # Stato UI throttled
        self._t_start: float = 0.0
        self._last_ui_update: float = 0.0
        self._display_handle: Any = None
        self._use_html_progress: bool = False

    def _on_training_start(self) -> None:
        self._t_start = time.perf_counter()
        self._last_ui_update = self._t_start

        if self.show_progress and self.total_timesteps > 0:
            try:
                from IPython import get_ipython
                from IPython.display import HTML, display
                shell = get_ipython()
                if shell is not None and shell.__class__.__name__ in (
                    "ZMQInteractiveShell",
                    "Shell",  # Google Colab
                ):
                    self._use_html_progress = True
                    self._display_handle = display(
                        HTML(self._render_html_progress(0, 0.0, 0.0)),
                        display_id=True,
                    )
            except Exception:
                self._use_html_progress = False

    def _on_step(self) -> bool:
        infos = self.locals.get("infos")
        if not infos:
            return True

        for info in infos:
            ep_info = info.get("episode")
            if ep_info is None:
                continue

            ep_r = float(ep_info["r"])
            ep_l = int(ep_info["l"])
            self.episode_rewards.append(ep_r)
            self.episode_lengths.append(ep_l)

            is_succ = bool(info.get("is_success", False))
            if is_succ:
                self.outcomes.append("success")
                self.successes += 1
                succ_flag = 1
            elif info.get("is_crashed", False):
                self.outcomes.append("crash")
                self.crashes += 1
                succ_flag = 0
            elif info.get("is_timeout", False):
                self.outcomes.append("timeout")
                self.timeouts += 1
                succ_flag = 0
            else:
                self.outcomes.append("unknown")
                succ_flag = 0

            tel = info.get("telemetry")
            if tel:
                fuel = float(tel["fuel_used"])
                self.fuel_used_list.append(fuel)
                self.air_steps_list.append(int(tel["air_steps"]))
                self.ground_steps_list.append(int(tel["ground_steps"]))
                self.final_x_list.append(float(tel["final_abs_x"]))
                self.final_vy_list.append(float(tel["final_abs_vy"]))
                self.final_angle_list.append(float(tel["final_abs_angle"]))
            else:
                fuel = 0.0
                self.fuel_used_list.append(0.0)
                self.air_steps_list.append(ep_l)
                self.ground_steps_list.append(0)
                self.final_x_list.append(0.0)
                self.final_vy_list.append(0.0)
                self.final_angle_list.append(0.0)

            # Aggiornamento O(1) della media mobile
            if len(self._window_outcomes) == self.window_size:
                self._window_successes -= self._window_outcomes[0]
            self._window_outcomes.append(succ_flag)
            self._window_successes += succ_flag
            self._window_fuels.append(fuel)

            # Snapshot in RAM del miglior modello (zero disk I/O!)
            if self.best_model_save_dir and len(self._window_outcomes) == self.window_size:
                curr_rate = (self._window_successes / self.window_size) * 100.0
                curr_fuel = sum(self._window_fuels) / self.window_size
                is_better = (
                    curr_rate >= 40.0
                    and (
                        curr_rate > self.best_moving_success_rate
                        or (
                            curr_rate == self.best_moving_success_rate
                            and curr_fuel < self._best_moving_fuel - 2.0
                        )
                    )
                )
                if is_better:
                    self.best_moving_success_rate = curr_rate
                    self._best_moving_fuel = curr_fuel
                    self._best_policy_state = {
                        k: v.detach().cpu().clone()
                        for k, v in self.model.policy.state_dict().items()
                    }

        # Snapshot intermedi per timelapse evoluzione (RAM, eseguito 3 volte durante tutto il training)
        if self.capture_milestones and self.total_timesteps > 0 and self._next_milestone_idx < len(self._milestone_targets):
            target_step = int(self._milestone_targets[self._next_milestone_idx] * self.total_timesteps)
            if self.num_timesteps >= target_step:
                pct = int(self._milestone_targets[self._next_milestone_idx] * 100)
                self.milestone_snapshots.append({
                    "step": self.num_timesteps,
                    "percent": pct,
                    "success_rate": self.current_moving_success_rate,
                    "state_dict": {
                        k: v.detach().cpu().clone()
                        for k, v in self.model.policy.state_dict().items()
                    },
                })
                self._next_milestone_idx += 1

        # Aggiornamento UI throttled (solo per tempo trascorso, non per step)
        if self.show_progress and self.total_timesteps > 0:
            now = time.perf_counter()
            if (now - self._last_ui_update) >= self.progress_interval_sec:
                self._last_ui_update = now
                self._update_progress_ui(now)

        return True

    def _on_training_end(self) -> None:
        if self.capture_milestones and hasattr(self, "model") and hasattr(self.model, "policy"):
            self.milestone_snapshots.append({
                "step": max(1, self.num_timesteps),
                "percent": 100,
                "success_rate": self.current_moving_success_rate,
                "state_dict": {
                    k: v.detach().cpu().clone()
                    for k, v in self.model.policy.state_dict().items()
                },
            })

        if self.show_progress and self.total_timesteps > 0:
            self._update_progress_ui(time.perf_counter(), finished=True)

    @property
    def current_moving_success_rate(self) -> float:
        if not self._window_outcomes:
            return 0.0
        return (self._window_successes / len(self._window_outcomes)) * 100.0

    def flush_best_model_to_disk(
        self,
        keep_best_loaded_if_superior: bool = True,
        superiority_margin: float = 5.0,
    ) -> bool:
        """
        Scrive su disco una sola volta il miglior modello catturato in RAM.
        Se `keep_best_loaded_if_superior=True` e il picco in training supera la
        media mobile finale di almeno `superiority_margin` punti percentuali,
        mantiene i pesi migliori caricati nell'agente per la valutazione.

        Restituisce True se i pesi migliori sono rimasti attivi in `self.model`.
        """
        if self._best_policy_state is None or self.best_model_save_dir is None:
            return False

        self.best_model_save_dir.mkdir(parents=True, exist_ok=True)
        save_p = self.best_model_save_dir / "best_model.zip"

        final_state = {
            k: v.detach().cpu().clone()
            for k, v in self.model.policy.state_dict().items()
        }

        # Carica pesi migliori e salva una sola volta su disco
        self.model.policy.load_state_dict(self._best_policy_state)
        self.model.save(str(save_p))
        self.best_model_path = save_p

        final_rate = self.current_moving_success_rate
        should_keep_best = (
            keep_best_loaded_if_superior
            and (self.best_moving_success_rate >= final_rate + superiority_margin)
        )

        if not should_keep_best:
            self.model.policy.load_state_dict(final_state)
            return False

        return True

    def _render_html_progress(
        self,
        steps: int,
        fps: float,
        eta_sec: float,
        finished: bool = False,
    ) -> str:
        pct = min(100.0, (steps / max(1, self.total_timesteps)) * 100.0)
        succ_rate = self.current_moving_success_rate
        mean_fuel = (
            sum(self._window_fuels) / len(self._window_fuels)
            if self._window_fuels
            else 0.0
        )
        bar_color = "#2e7d32" if finished or succ_rate >= 80.0 else "#1976d2"
        status_txt = "Completato" if finished else f"ETA: {eta_sec:.0f}s"

        return (
            f'<div style="font-family:sans-serif; background:#f8f9fa; border:1px solid #dee2e6; '
            f'border-radius:8px; padding:10px 14px; margin:8px 0; max-width:760px;">'
            f'<div style="display:flex; justify-content:space-between; font-size:13px; margin-bottom:6px;">'
            f'<span><strong>Addestramento PPO:</strong> {steps:,} / {self.total_timesteps:,} step ({pct:.1f}%)</span>'
            f'<span><strong>{fps:,.0f} step/s</strong> | {status_txt}</span>'
            f'</div>'
            f'<div style="background:#e9ecef; border-radius:6px; height:12px; overflow:hidden;">'
            f'<div style="width:{pct:.1f}%; background:{bar_color}; height:100%; transition:width 0.25s;"></div>'
            f'</div>'
            f'<div style="display:flex; justify-content:space-between; font-size:12px; color:#495057; margin-top:6px;">'
            f'<span>Episodi: {len(self.outcomes)} | Successo recente: <strong>{succ_rate:.1f}%</strong> '
            f'(Picco: {max(0.0, self.best_moving_success_rate):.1f}%)</span>'
            f'<span>Carburante recente: {mean_fuel:.1f} un.</span>'
            f'</div></div>'
        )

    def _update_progress_ui(self, now: float, finished: bool = False) -> None:
        elapsed = max(1e-3, now - self._t_start)
        steps = self.num_timesteps
        fps = steps / elapsed
        remaining_steps = max(0, self.total_timesteps - steps)
        eta_sec = remaining_steps / max(1.0, fps)

        if self._use_html_progress and self._display_handle is not None:
            from IPython.display import HTML
            self._display_handle.update(
                HTML(self._render_html_progress(steps, fps, eta_sec, finished=finished))
            )
        else:
            pct = min(100.0, (steps / max(1, self.total_timesteps)) * 100.0)
            succ_rate = self.current_moving_success_rate
            end_char = "\n" if finished else ""
            print(
                f"\r[play2learn] {pct:5.1f}% | {steps:,}/{self.total_timesteps:,} step | "
                f"{fps:,.0f} step/s | Successo({self.window_size}ep): {succ_rate:5.1f}% | "
                f"ETA: {eta_sec:4.0f}s   ",
                end=end_char,
                flush=True,
            )


def create_ppo_agent(
    env: VecEnv,
    seed: int = 42,
    device: str = "cpu",
    custom_hyperparams: dict[str, Any] | None = None,
    verbose: int = 0,
    use_lr_schedule: bool = True,
) -> PPO:
    """
    Istanzia un agente PPO pre-configurato per convergenza rapida su LunarLander-v3.
    Adatta automaticamente `n_steps` al numero di ambienti paralleli per mantenere
    costante il rollout buffer target (2048 step, multiplo esatto di batch_size=64).
    """
    params = dict(DEFAULT_PPO_PARAMS)
    if custom_hyperparams:
        params.update(custom_hyperparams)

    # Adattamento automatico di n_steps a n_envs se non forzato manualmente
    if not (custom_hyperparams and "n_steps" in custom_hyperparams):
        n_envs = max(1, getattr(env, "num_envs", 4))
        batch_size = int(params.get("batch_size", 64))
        raw_steps = max(batch_size, TARGET_ROLLOUT_SIZE // n_envs)
        # Allinea al multiplo di batch_size per evitare minibatch troncati
        params["n_steps"] = max(batch_size, (raw_steps // batch_size) * batch_size)

    # Applicazione learning rate schedule lineare
    if use_lr_schedule and not callable(params.get("learning_rate")):
        lr_start = float(params.get("learning_rate", DEFAULT_LR_START))
        lr_end = min(DEFAULT_LR_END, lr_start * 0.1)
        params["learning_rate"] = LinearSchedule(lr_start, lr_end)

    return PPO(
        env=env,
        seed=seed,
        device=device,
        verbose=verbose,
        **params,
    )

"""
Modulo di valutazione e registrazione video/GIF ad alte prestazioni.
Procedura in 2 fasi:
- Fase 1A: Fast Sweep deterministico (inferenza diretta PyTorch, metriche imparziali).
- Fase 1B: Caccia stocastica opzionale ai soli candidati video mancanti (senza alterare i KPI).
- Fase 2: Replay fedele su Box2D dei soli target selezionati, encoding MP4 ultrafast
  e generazione GIF diretta dai frame in RAM (zero re-decoding da disco).
"""

from __future__ import annotations

import time
from pathlib import Path
from typing import Callable, Any

import numpy as np
import torch

try:
    import gymnasium as gym
    from stable_baselines3 import PPO
except ImportError:
    gym = Any  # type: ignore
    PPO = Any  # type: ignore

from play2learn_lunarlander.environment import CustomRewardWrapper


def _make_fast_predictor(model: PPO) -> Callable[[np.ndarray, bool], int]:
    """
    Crea un predictor leggero che bypassa le validazioni per-step di SB3 `model.predict()`,
    accelerando lo sweep CPU di 3-4x.
    """
    policy = getattr(model, "policy", None)
    if policy is None:
        return lambda obs, det: int(model.predict(obs, deterministic=det)[0])

    policy.set_training_mode(False)
    device = model.device

    @torch.inference_mode()
    def _predict(obs: np.ndarray, deterministic: bool) -> int:
        obs_t = torch.as_tensor(obs, dtype=torch.float32, device=device).unsqueeze(0)
        act_t = policy._predict(obs_t, deterministic=deterministic)
        return int(act_t.item())

    return _predict


def _run_single_sweep_episode(
    env: gym.Env,
    predict_fn: Callable[[np.ndarray, bool], int],
    ep_seed: int,
    deterministic: bool,
    t_start: float,
    timeout_sec: float,
) -> tuple[dict[str, Any], bool]:
    """
    Esegue un singolo episodio completo di sweep.
    Se il timer scatta durante il volo, completa comunque l'episodio corrente.
    """
    obs, _ = env.reset(seed=ep_seed)
    done = False
    ep_rew = 0.0
    steps = 0
    ep_actions: list[int] = []
    last_info: dict[str, Any] = {}
    timeout_hit = False

    while not done:
        act_int = predict_fn(obs, deterministic)
        ep_actions.append(act_int)
        obs, rew, terminated, truncated, last_info = env.step(act_int)
        ep_rew += float(rew)
        steps += 1
        done = terminated or truncated

        if not timeout_hit and (time.perf_counter() - t_start >= timeout_sec):
            timeout_hit = True

    tel = last_info.get("telemetry", {})
    fuel = float(tel.get("fuel_used", 0.0))

    candidate = {
        "seed": ep_seed,
        "actions": ep_actions,
        "is_success": bool(last_info.get("is_success", False)),
        "is_crashed": bool(last_info.get("is_crashed", False)),
        "is_timeout": bool(last_info.get("is_timeout", False)),
        "is_off_target": bool(last_info.get("is_off_target", False)),
        "steps": steps,
        "fuel": fuel,
        "reward": ep_rew,
        "deterministic": deterministic,
    }
    return candidate, timeout_hit


def evaluate_and_record(
    model: PPO,
    env_id: str,
    reward_fn: Callable,
    video_dir: Path | str,
    target_successes: int = 3,
    target_failures: int = 3,
    sweep_timeout_seconds: float = 6.0,
    min_eval_episodes: int = 20,
    max_eval_episodes: int = 80,
    seed: int = 100,
    env_kwargs: dict[str, Any] | None = None,
    create_gif: bool = True,
    filename_prefix: str = "",
) -> dict[str, Any]:
    """
    Valuta l'agente in modo statisticamente imparziale (policy deterministica)
    e registra fino a `target_successes` successi e `target_failures` fallimenti.
    """
    env_kwargs = env_kwargs or {}
    video_dir = Path(video_dir)
    video_dir.mkdir(parents=True, exist_ok=True)

    print(
        f"\n[play2learn] Valutazione dell'agente: ricerca di {target_successes} successi "
        f"e {target_failures} fallimenti..."
    )

    predict_fn = _make_fast_predictor(model)
    try:
        sweep_env = gym.make(env_id, render_mode=None, disable_env_checker=True, **env_kwargs)
    except TypeError:
        sweep_env = gym.make(env_id, render_mode=None, **env_kwargs)
    sweep_env = CustomRewardWrapper(sweep_env, reward_fn)

    success_candidates: list[dict[str, Any]] = []
    failure_candidates: list[dict[str, Any]] = []

    # Metriche ufficiali: calcolate SOLO sugli episodi deterministici (zero bias!)
    eval_rewards: list[float] = []
    eval_lengths: list[int] = []
    eval_fuels: list[float] = []
    success_fuels: list[float] = []
    failure_fuels: list[float] = []
    eval_successes: int = 0
    eval_crashes: int = 0
    eval_timeouts: int = 0

    t_sweep_start = time.perf_counter()
    timeout_triggered = False
    ep = 0

    # =========================================================================
    # FASE 1A: SWEEP DETERMINISTICO IMPARZIALE
    # =========================================================================
    det_budget_sec = sweep_timeout_seconds * 0.75
    while ep < max_eval_episodes and not timeout_triggered:
        have_enough_videos = (
            len(success_candidates) >= target_successes
            and len(failure_candidates) >= target_failures
        )
        if ep >= min_eval_episodes and (
            have_enough_videos or (time.perf_counter() - t_sweep_start >= det_budget_sec)
        ):
            break

        cand, timeout_triggered = _run_single_sweep_episode(
            env=sweep_env,
            predict_fn=predict_fn,
            ep_seed=seed + ep,
            deterministic=True,
            t_start=t_sweep_start,
            timeout_sec=sweep_timeout_seconds,
        )

        eval_rewards.append(cand["reward"])
        eval_lengths.append(cand["steps"])
        eval_fuels.append(cand["fuel"])

        if cand["is_success"]:
            eval_successes += 1
            success_fuels.append(cand["fuel"])
            success_candidates.append(cand)
        else:
            if cand["is_timeout"]:
                eval_timeouts += 1
            else:
                eval_crashes += 1
            failure_fuels.append(cand["fuel"])
            failure_candidates.append(cand)

        ep += 1

    # =========================================================================
    # FASE 1B: CACCIA RAPIDA STOCASTICA AI SOLI VIDEO MANCANTI (NON ALTERA I KPI)
    # =========================================================================
    stoch_ep = 0
    max_stoch_hunt = 25
    while (
        not timeout_triggered
        and stoch_ep < max_stoch_hunt
        and (
            len(success_candidates) < target_successes
            or len(failure_candidates) < target_failures
        )
    ):
        cand, timeout_triggered = _run_single_sweep_episode(
            env=sweep_env,
            predict_fn=predict_fn,
            ep_seed=seed + 5000 + stoch_ep,
            deterministic=False,
            t_start=t_sweep_start,
            timeout_sec=sweep_timeout_seconds,
        )
        if cand["is_success"] and len(success_candidates) < target_successes:
            success_candidates.append(cand)
        elif (not cand["is_success"]) and len(failure_candidates) < target_failures:
            failure_candidates.append(cand)
        stoch_ep += 1

    sweep_env.close()
    sweep_duration = time.perf_counter() - t_sweep_start
    total_evaluated = max(1, len(eval_rewards))
    success_rate = (eval_successes / total_evaluated) * 100.0

    print(
        f"[play2learn] Sweep completato in {sweep_duration:.2f}s "
        f"({total_evaluated} episodi deterministici + {stoch_ep} ricerca video): "
        f"{len(success_candidates)} successi, {len(failure_candidates)} fallimenti pronti."
    )

    # =========================================================================
    # FASE 2: REPLAY VIDEO MIRATO (SENZA WRAPPER, GIF DA FRAME IN RAM)
    # =========================================================================
    # Priorità ai candidati deterministici, ordinati per efficienza carburante
    selected_successes = sorted(
        success_candidates,
        key=lambda c: (not c["deterministic"], c["fuel"]),
    )[:target_successes]

    # Bilancia crash e timeout tra i fallimenti se entrambi presenti
    crashes_pool = [c for c in failure_candidates if c["is_crashed"]]
    timeouts_pool = [c for c in failure_candidates if c["is_timeout"]]
    selected_failures: list[dict[str, Any]] = []
    if crashes_pool and timeouts_pool and target_failures >= 2:
        selected_failures.extend(crashes_pool[: target_failures - 1])
        selected_failures.append(timeouts_pool[0])
    for c in failure_candidates:
        if len(selected_failures) >= target_failures:
            break
        if c not in selected_failures:
            selected_failures.append(c)

    success_videos: list[Path] = []
    failure_videos: list[Path] = []
    best_success_fuel: float = float("inf")
    best_success_video: Path | None = None
    best_success_frames: list[np.ndarray] | None = None

    to_render = (
        [("successo", c) for c in selected_successes]
        + [("fallimento", c) for c in selected_failures]
    )

    if to_render:
        print(
            f"[play2learn] Fase 2: Rendering video ({len(selected_successes)} successi, "
            f"{len(selected_failures)} fallimenti)..."
        )
        try:
            render_env = gym.make(
                env_id,
                render_mode="rgb_array",
                disable_env_checker=True,
                **env_kwargs,
            )
        except TypeError:
            render_env = gym.make(env_id, render_mode="rgb_array", **env_kwargs)

        for kind, cand in to_render:
            render_env.reset(seed=cand["seed"])
            frames: list[np.ndarray] = []
            init_frame = render_env.render()
            if init_frame is not None:
                frames.append(init_frame)

            for act in cand["actions"]:
                _, _, term, trunc, _ = render_env.step(act)
                frame = render_env.render()
                if frame is not None:
                    frames.append(frame)
                if term or trunc:
                    break

            pfx = f"{filename_prefix}_" if filename_prefix else ""
            if kind == "successo":
                idx = len(success_videos) + 1
                v_path = video_dir / f"{pfx}successo_{idx}_step_{cand['steps']}.mp4"
                _save_frames_as_mp4(frames, v_path)
                success_videos.append(v_path)
                if cand["fuel"] < best_success_fuel:
                    best_success_fuel = cand["fuel"]
                    best_success_video = v_path
                    best_success_frames = frames
            else:
                idx = len(failure_videos) + 1
                if cand.get("is_off_target"):
                    causa = "fuori_bersaglio"
                elif cand["is_crashed"]:
                    causa = "crash"
                elif cand["is_timeout"]:
                    causa = "timeout"
                else:
                    causa = "fallimento"
                v_path = video_dir / f"{pfx}fallimento_{idx}_{causa}_step_{cand['steps']}.mp4"
                _save_frames_as_mp4(frames, v_path)
                failure_videos.append(v_path)

        render_env.close()

    # Genera GIF direttamente dai frame già in RAM (evita decode H.264 da disco)
    gif_path: Path | None = None
    if create_gif and best_success_video and best_success_frames:
        gif_path = _save_frames_as_gif(
            best_success_frames,
            best_success_video.with_suffix(".gif"),
        )

    return {
        "mean_reward": float(np.mean(eval_rewards)) if eval_rewards else 0.0,
        "std_reward": float(np.std(eval_rewards)) if eval_rewards else 0.0,
        "mean_length": float(np.mean(eval_lengths)) if eval_lengths else 0.0,
        "mean_fuel": float(np.mean(eval_fuels)) if eval_fuels else 0.0,
        "eval_success_rate": success_rate,
        "total_evaluated_episodes": total_evaluated,
        "eval_successes": eval_successes,
        "eval_crashes": eval_crashes,
        "eval_timeouts": eval_timeouts,
        "success_videos": success_videos,
        "failure_videos": failure_videos,
        "videos": success_videos + failure_videos,
        "gif_path": gif_path,
        "mean_fuel_success": float(np.mean(success_fuels)) if success_fuels else 0.0,
        "mean_fuel_failure": float(np.mean(failure_fuels)) if failure_fuels else 0.0,
    }


def _save_frames_as_mp4(frames: list[np.ndarray], output_path: Path, fps: int = 50) -> None:
    """Salva i frame in MP4 usando preset ultrafast (2-3x più veloce e file compatti)."""
    try:
        import imageio
        output_path.parent.mkdir(parents=True, exist_ok=True)
        imageio.mimsave(
            str(output_path),
            frames,
            fps=fps,
            macro_block_size=1,
            ffmpeg_params=["-preset", "ultrafast", "-crf", "26"],
        )
    except Exception:
        # Fallback compatibile se il backend ffmpeg non accetta ffmpeg_params
        try:
            import imageio
            imageio.mimsave(str(output_path), frames, fps=fps, macro_block_size=1)
        except Exception as e:
            print(f"[play2learn] Errore nel salvataggio video {output_path.name}: {e}")


def _save_frames_as_gif(
    frames: list[np.ndarray],
    gif_output: Path,
    source_fps: int = 50,
    stride: int = 2,
) -> Path | None:
    """Crea la GIF animata direttamente dai frame in RAM campionati 1 ogni `stride`."""
    try:
        import imageio
        sampled = frames[::stride]
        if sampled:
            imageio.mimsave(
                str(gif_output),
                sampled,
                fps=max(10, source_fps // stride),
                loop=0,
            )
            return gif_output
    except Exception:
        pass
    return None


def crea_gif_evoluzione_apprendimento(
    model: PPO,
    milestones: list[dict[str, Any]],
    env_id: str,
    output_path: Path | str,
    env_kwargs: dict[str, Any] | None = None,
    seed: int = 42,
    max_steps_per_episode: int = 240,
    stride: int = 2,
    target_width: int = 400,
    fps: int = 15,
) -> Path | None:
    """
    Genera un timelapse GIF fluido e annotato che mostra visivamente come l'agente
    è passato dall'esplorazione iniziale (schianti) all'atterraggio morbido,
    confrontando le milestone di training nel medesimo scenario lunare.
    """
    if not milestones:
        return None

    output_path = Path(output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    env_kwargs = env_kwargs or {}

    try:
        from PIL import Image, ImageDraw
        import imageio
        import gymnasium as gym
    except ImportError as e:
        print(f"[play2learn] Modulo mancante per la GIF di evoluzione: {e}")
        return None

    policy = getattr(model, "policy", None)
    if policy is None:
        return None

    # Salva stato originale dei pesi del modello per ripristinarlo intatto
    orig_state = {
        k: v.detach().cpu().clone()
        for k, v in policy.state_dict().items()
    }

    try:
        render_env = gym.make(
            env_id,
            render_mode="rgb_array",
            disable_env_checker=True,
            **env_kwargs,
        )
    except TypeError:
        render_env = gym.make(env_id, render_mode="rgb_array", **env_kwargs)

    all_annotated_frames: list[np.ndarray] = []
    total_stages = len(milestones)

    fasi_titoli = [
        "Fase 1: Primi Passi (Esplorazione Casuale)",
        "Fase 2: Accensione Motori (Tentativi Stabilizzazione)",
        "Fase 3: Discesa Orientata (Controllo Quota)",
        "Fase Finale: Atterraggio Controllato",
    ]

    try:
        for idx, m in enumerate(milestones):
            policy.load_state_dict(m["state_dict"])
            predict_fn = _make_fast_predictor(model)

            # Usa sempre lo stesso seed per visualizzare la progressione sullo stesso identico terreno
            obs, _ = render_env.reset(seed=seed)
            ep_frames: list[np.ndarray] = []
            f_init = render_env.render()
            if f_init is not None:
                ep_frames.append(f_init)

            done = False
            step_count = 0
            final_esito = "In volo..."

            while not done and step_count < max_steps_per_episode:
                action = predict_fn(obs, deterministic=True)
                obs, rew, term, trunc, info = render_env.step(action)
                step_count += 1
                done = term or trunc

                f = render_env.render()
                if f is not None:
                    ep_frames.append(f)

                if done:
                    if info.get("is_success", False):
                        final_esito = "🎯 Allunaggio Riuscito!"
                    elif info.get("is_crashed", False):
                        final_esito = "💥 Schianto al suolo"
                    elif info.get("is_timeout", False):
                        final_esito = "⏳ Tempo scaduto (Hovering)"
                    else:
                        final_esito = "Fine volo"

            stage_title = fasi_titoli[idx] if idx < len(fasi_titoli) else f"Fase {idx+1}/{total_stages}"
            step_val = m.get("step", step_count)
            pct_val = m.get("percent", int((idx + 1) / total_stages * 100))
            succ_val = m.get("success_rate", 0.0)

            sampled_ep = ep_frames[::stride]
            if not sampled_ep:
                continue

            # Congela il frame finale per 8 frame per rendere leggibile l'esito
            if done and sampled_ep:
                sampled_ep = sampled_ep + [sampled_ep[-1]] * 8

            for f in sampled_ep:
                h, w = f.shape[:2]
                target_h = int(target_width * h / w)
                img = Image.fromarray(f).resize((target_width, target_h), Image.Resampling.BILINEAR)
                draw = ImageDraw.Draw(img)

                draw.rectangle([(0, 0), (target_width, 34)], fill=(15, 23, 42))
                draw.text((10, 4), f"{stage_title} — Step ~{step_val:,} ({pct_val}%)", fill=(255, 255, 255))
                draw.text((10, 18), f"Esito: {final_esito} | Successo mobile: {succ_val:.0f}%", fill=(147, 197, 253))

                all_annotated_frames.append(np.array(img))

    finally:
        render_env.close()
        # Ripristina invariati i pesi del modello originale
        policy.load_state_dict(orig_state)

    if not all_annotated_frames:
        return None

    try:
        imageio.mimsave(str(output_path), all_annotated_frames, fps=fps, loop=0)
        return output_path
    except Exception as e:
        print(f"[play2learn] Errore nel salvataggio della GIF evoluzione: {e}")
        return None

"""
Modulo 'Flight Coach' - Diagnostica didattica intelligente del comportamento dell'agente.
Fornisce spiegazioni logico-matematiche chiare sulle anomalie di apprendimento (Reward Hacking,
trappole di hovering o kamikaze) e suggerisce come correggere la formula di reward.
"""

from __future__ import annotations
from typing import Any
import numpy as np

from play2learn_lunarlander.agent import RewardTrackerCallback


def diagnose_agent_behavior(tracker: RewardTrackerCallback) -> dict[str, Any]:
    """
    Analizza gli ultimi episodi completati dall'agente per rilevare eventuali
    comportamenti anomali o subottimali e formulare consigli didattici mirati.

    Restituisce un dizionario strutturato con metriche e testo descrittivo.
    """
    n = len(tracker.outcomes)
    if n == 0:
        return {
            "status": "nessun_dato",
            "title": "Nessun dato disponibile",
            "message": "Nessun episodio completato.",
            "suggestion": "Verifica che l'addestramento non sia stato interrotto subito.",
            "formatted_text": "Nessun episodio completato per la diagnostica.",
            "metrics": {},
        }

    tail = min(50, n)
    recent_outcomes = tracker.outcomes[-tail:]
    succ_rate = (recent_outcomes.count("success") / tail) * 100.0
    crash_rate = (recent_outcomes.count("crash") / tail) * 100.0
    time_rate = (recent_outcomes.count("timeout") / tail) * 100.0

    mean_len = float(np.mean(tracker.episode_lengths[-tail:]))
    mean_fuel = float(np.mean(tracker.fuel_used_list[-tail:]))
    mean_ground = float(np.mean(tracker.ground_steps_list[-tail:]))
    mean_x = float(np.mean(tracker.final_x_list[-tail:]))
    mean_vy = float(np.mean(tracker.final_vy_list[-tail:]))
    mean_angle = float(np.mean(tracker.final_angle_list[-tail:])) if tracker.final_angle_list else 0.0

    metrics = {
        "tail": tail,
        "success_rate": succ_rate,
        "crash_rate": crash_rate,
        "timeout_rate": time_rate,
        "mean_length": mean_len,
        "mean_fuel": mean_fuel,
        "mean_ground_steps": mean_ground,
        "mean_abs_x": mean_x,
        "mean_abs_vy": mean_vy,
        "mean_abs_angle": mean_angle,
    }

    # Determinazione diagnosi principale
    if succ_rate >= 75.0:
        status = "ottimo"
        title = "ALLUNAGGIO PERFETTO! 🎯"
        msg = "L'agente ha imparato ad atterrare con sicurezza e centrare la piazzola."
        sugg = "Ottimo lavoro! Puoi provare ora ad ottimizzare il consumo di carburante o passare a una missione con vento/gravità estrema."
    elif time_rate >= 40.0 and mean_ground > 80:
        status = "motore_acceso_a_terra"
        title = "TRAPPOLA: MOTORI ACCESI A TERRA 🔥"
        msg = "L'agente tocca la piazzola ma non spegne i motori, continuando a consumare carburante fino al timeout."
        sugg = "Quando entrambe le zampe toccano terra (left_leg==1 e right_leg==1), dai un premio speciale all'azione 0 (action == 0)."
    elif time_rate >= 40.0:
        status = "hovering_infinito"
        title = "TRAPPOLA: HOVERING INFINITO (PAURA DI SCENDERE) 🪂"
        msg = "L'agente galleggia in aria a mezza altezza per evitare di schiantarsi, consumando tutti i 1000 step."
        sugg = "Riduci o rimuovi qualsiasi premio positivo assegnato 'in volo'; penalizza l'altitudine 'abs(y)' e dai una forte punizione a 'is_timeout'."
    elif crash_rate >= 70.0 and mean_len < 95:
        status = "kamikaze"
        title = "TRAPPOLA: STRATEGIA KAMIKAZE 💥"
        msg = "L'agente si schianta deliberatamente al suolo in pochi istanti per interrompere subito l'episodio."
        sugg = "Le penalità che assegni ad ogni step in volo sono troppo severe: all'agente conviene morire subito piuttosto che accumulare penalità continue! Riduci le penalità di step o aumenta la punizione per 'is_crashed'."
    elif crash_rate >= 50.0 and mean_vy > 0.35:
        status = "impatto_veloce"
        title = "IMPATTO TROPPO VIOLENTO 🏎️"
        msg = "L'agente scende dritto sulla piazzola ma con una velocità verticale eccessiva (|vy| > 0.35), distruggendo le zampe."
        sugg = "Aumenta la penalità sulla velocità di discesa abs(vy), specialmente alle quote basse (ad esempio quando y < 0.5 moltiplica la penalità per 1.5 - y)."
    elif crash_rate >= 50.0 and mean_x > 0.25:
        status = "fuori_bersaglio"
        title = "ATTERRAGGIO FUORI BERSAGLIO 📍"
        msg = "L'agente tocca il terreno roccioso fuori dalle bandierine di segnalazione (|x| > 0.20)."
        sugg = "Aumenta la penalità sulla distanza orizzontale abs(x) e sulla velocità orizzontale abs(vx) per forzare il modulo verso il centro x = 0.0."
    elif crash_rate >= 50.0 and mean_angle > 0.25:
        status = "assetto_inclinato"
        title = "ASSETTO TROPPO INCLINATO 📐"
        msg = "L'astronave si rovescia prima del contatto a causa di un'inclinazione eccessiva (|angle| > 0.25 rad)."
        sugg = "Aumenta il coefficiente negativo su abs(angle) e su abs(vangle) per stabilizzare la verticale."
    else:
        status = "in_apprendimento"
        title = "APPRENDIMENTO IN CORSO ⚙️"
        msg = "L'agente sta iniziando a comprendere la dinamica, ma la combinazione dei pesi è ancora instabile."
        sugg = "Controlla l'equilibrio tra i termini: dai priorità prima alla stabilizzazione dell'angolo e della discesa verticale, poi alla centratura orizzontale."

    lines = [
        f"ANALISI DEGLI ULTIMI {tail} EPISODI:",
        f"• Successi: {succ_rate:.1f}% | Schianti: {crash_rate:.1f}% | Timeout: {time_rate:.1f}%",
        f"• Durata media: {mean_len:.0f} step ({mean_len/50:.1f}s) | Carburante medio: {mean_fuel:.1f} unità",
        f"• Posizione finale media: |x| = {mean_x:.2f} | Velocità impatto: |vy| = {mean_vy:.2f}",
        f"\nDIAGNOSI FLIGHT COACH: {title}",
        f"{msg}",
        f"💡 CONSIGLIO: {sugg}",
    ]

    return {
        "status": status,
        "title": title,
        "message": msg,
        "suggestion": sugg,
        "formatted_text": "\n".join(lines),
        "metrics": metrics,
    }

"""
Modulo di visualizzazione grafica: mappa dell'ambiente, dashboard telemetrica 2x2,

from __future__ import annotations

import base64

from typing import Any

from matplotlib.figure import Figure

def mostra_mappa_ambiente() -> None:
    """
    Genera e visualizza nel Notebook una scheda grafica annotata dell'ambiente
    LunarLander-v3 con sistema di riferimento, coordinate piazzola, limiti di rottura e motori.
    """
    try:
        import gymnasium as gym
    except ImportError as e:

        ) from e

    env = gym.make("LunarLander-v3", render_mode="rgb_array")

def plot_training_results(
    tracker: RewardTrackerCallback,
    mission_title: str,
    save_path: Path | str
) -> dict[str, Any]:
    """
    Genera e salva una dashboard 2x2 con:
    1. Custom Reward per episodio (curva istantanea e media mobile)
    2. Tasso di Successo / Schianto / Timeout (%) nel tempo
    3. Durata del volo (Step per episodio con fascia ideale 150-350)
    4. Consumo di Carburante (unità spese)
    Include in calce il box della Diagnosi Flight Coach.
    """
    save_path = Path(save_path)
    save_path.parent.mkdir(parents=True, exist_ok=True)

    n_episodes = len(tracker.episode_rewards)
    diag = diagnose_agent_behavior(tracker)
    diag_text = diag["formatted_text"]

    if n_episodes == 0:
        print("[play2learn] Nessun episodio completato per generare il grafico.")
        return diag

    episodes = np.arange(1, n_episodes + 1)
    window = min(40, max(5, n_episodes // 12))

    fig = Figure(figsize=(14, 8.8), dpi=300)
    FigureCanvasAgg(fig)
    fig.suptitle(f"Telemetria di Addestramento RL — {mission_title}", fontsize=14, fontweight="bold")
    axes = fig.subplots(2, 2)
    (ax1, ax2), (ax3, ax4) = axes

    # 1. Custom Reward
    ax1.plot(episodes, tracker.episode_rewards, color="#90caf9", alpha=0.45, label="Reward Episodio", rasterized=True)
    if n_episodes >= window:
        mov_rew = np.convolve(tracker.episode_rewards, np.ones(window) / window, mode="valid")
        ax1.plot(range(window, n_episodes + 1), mov_rew, color="#0d47a1", lw=2.2, label=f"Media Mobile ({window} ep)")
    ax1.set_title("1. Ricompensa Cumulata (La tua Custom Reward)", fontweight="bold")
    ax1.set_ylabel("Custom Reward")
    ax1.grid(True, linestyle="--", alpha=0.5)
    ax1.legend(loc="upper left", fontsize=8.5)

    # 2. Percentuali Esiti nel Tempo
    succ_arr = np.array([100.0 if o == "success" else 0.0 for o in tracker.outcomes])
    crash_arr = np.array([100.0 if o == "crash" else 0.0 for o in tracker.outcomes])
    time_arr = np.array([100.0 if o == "timeout" else 0.0 for o in tracker.outcomes])

    if n_episodes >= window:
        x_roll = range(window, n_episodes + 1)
        ax2.plot(x_roll, np.convolve(succ_arr, np.ones(window) / window, mode="valid"), color="#2e7d32", lw=2.2, label="% Allunaggi Riusciti")
        ax2.plot(x_roll, np.convolve(crash_arr, np.ones(window) / window, mode="valid"), color="#c62828", lw=1.8, linestyle="--", label="% Schianti")
        ax2.plot(x_roll, np.convolve(time_arr, np.ones(window) / window, mode="valid"), color="#ef6c00", lw=1.8, linestyle=":", label="% Timeout (Hovering)")
    else:
        ax2.plot(episodes, succ_arr, color="#2e7d32", label="% Successo")
    ax2.set_title("2. Esito delle Missioni nel Tempo (%)", fontweight="bold")
    ax2.set_ylabel("Percentuale (%)")
    ax2.set_ylim(-5, 105)
    ax2.grid(True, linestyle="--", alpha=0.5)
    ax2.legend(loc="center left", fontsize=8.5)

    # 3. Durata del Volo
    ax3.plot(episodes, tracker.episode_lengths, color="#a5d6a7", alpha=0.45, label="Step per Episodio", rasterized=True)
    if n_episodes >= window:
        mov_len = np.convolve(tracker.episode_lengths, np.ones(window) / window, mode="valid")
        ax3.plot(range(window, n_episodes + 1), mov_len, color="#1b5e20", lw=2.0, label=f"Media Mobile ({window} ep)")
    ax3.axhspan(150, 350, color="#00e676", alpha=0.12, label="Fascia durata ideale (150-350 step)")
    ax3.axhline(1000, color="#ef6c00", linestyle=":", lw=1.2, label="Limite Timeout (1000 step)")
    ax3.set_title("3. Durata del Volo (Step: <90 Kamikaze, 1000 Timeout)", fontweight="bold")
    ax3.set_xlabel("Numero di Episodio")
    ax3.set_ylabel("Numero di Step (50 step = 1s)")
    ax3.grid(True, linestyle="--", alpha=0.5)
    ax3.legend(loc="upper left", fontsize=8.5)

    # 4. Consumo di Carburante
    ax4.plot(episodes, tracker.fuel_used_list, color="#ffcc80", alpha=0.45, label="Carburante Speso", rasterized=True)
    if n_episodes >= window:
        mov_fuel = np.convolve(tracker.fuel_used_list, np.ones(window) / window, mode="valid")
        ax4.plot(range(window, n_episodes + 1), mov_fuel, color="#e65100", lw=2.0, label=f"Media Mobile ({window} ep)")
    ax4.set_title("4. Carburante Consumato (1.0 princ., 0.3 lat.)", fontweight="bold")
    ax4.set_xlabel("Numero di Episodio")
    ax4.set_ylabel("Unità di Carburante")
    ax4.grid(True, linestyle="--", alpha=0.5)
    ax4.legend(loc="upper left", fontsize=8.5)

    fig.tight_layout(rect=[0, 0.15, 1, 0.95])

    # Box Diagnosi Flight Coach posizionato in calce
    # Sostituisce eventuali emoji non presenti nei font standard per evitare warning grafici
    clean_diag = (
        diag_text.replace("🏎️", "")
        .replace("📍", "")
        .replace("📐", "")
        .replace("⚙️", "")
        .replace("💡", "-> ")
        .replace("💥", "*")
        .replace("🎯", "[OK]")
    )
    fig.text(
        0.5, 0.015, clean_diag, ha="center", va="bottom", fontsize=9.2, family="sans-serif",
        bbox=dict(boxstyle="round,pad=0.55", facecolor="#fffde7", edgecolor="#fbc02d", lw=1.6)
    )

    fig.savefig(str(save_path), dpi=300)
    fig.clear()
    return diag

def plot_comparison(runs_data: list[dict[str, Any]], save_path: Path | str) -> None:
    """
    Genera un grafico comparativo tra più run (utile per la reportistica tecnica:
    es. confronto tra approccio iniziale ingenuo vs approccio ingegnerizzato con reward shaping).
    """
    save_path = Path(save_path)
    save_path.parent.mkdir(parents=True, exist_ok=True)

    if not runs_data:
        return

    n_runs = len(runs_data)
    labels = []
    for i, r in enumerate(runs_data):
        base_name = r.get("name", f"Run {i+1}")
        ts = r.get("timestamp")
        if ts and ts not in base_name:
            labels.append(f"{base_name}\n({ts})")
        else:
            labels.append(base_name)
    success_rates = [float(r.get("eval_success_rate", 0.0)) for r in runs_data]
    fuel_means = [float(r.get("eval_mean_fuel", 0.0)) for r in runs_data]
    lengths = [float(r.get("eval_mean_length", 0.0)) for r in runs_data]

    x = np.arange(n_runs)
    width = 0.28

    fig = Figure(figsize=(13, 5), dpi=300)
    FigureCanvasAgg(fig)
    fig.suptitle("Confronto Esperimenti di Reinforcement Learning — Report Tecnico", fontsize=13, fontweight="bold")
    (ax1, ax2) = fig.subplots(1, 2)

    # Grafico 1: Tasso di Successo
    bars1 = ax1.bar(x, success_rates, width, color="#2e7d32", alpha=0.85, edgecolor="black")
    ax1.set_title("Tasso di Allunaggio Riuscito (%)", fontweight="bold")
    ax1.set_ylabel("Success Rate (%)")
    ax1.set_ylim(0, 110)
    ax1.set_xticks(x)
    ax1.set_xticklabels(labels, rotation=15, ha="right")
    ax1.grid(axis="y", linestyle="--", alpha=0.5)

    for bar in bars1:
        yval = bar.get_height()
        ax1.text(bar.get_x() + bar.get_width()/2.0, yval + 2, f"{yval:.0f}%", ha="center", va="bottom", fontweight="bold")

    # Grafico 2: Consumo di Carburante e Durata
    bars2 = ax2.bar(x - width/2, fuel_means, width, label="Carburante Medio", color="#ef6c00", alpha=0.85, edgecolor="black")
    bars3 = ax2.bar(x + width/2, [l/5 for l in lengths], width, label="Durata (step / 5)", color="#1565c0", alpha=0.85, edgecolor="black")
    ax2.set_title("Efficienza del Volo (Consumo e Durata)", fontweight="bold")
    ax2.set_xticks(x)
    ax2.set_xticklabels(labels, rotation=15, ha="right")
    ax2.grid(axis="y", linestyle="--", alpha=0.5)
    ax2.legend(loc="upper right")

    fig.tight_layout()
    fig.savefig(str(save_path), dpi=300)
    fig.clear()

"""
Modulo per la gestione della persistenza, salvataggio dei checkpoint dei modelli,
delle metriche, del codice della funzione di reward e dell'alberatura 'runs/'.
"""

from __future__ import annotations

import json
import inspect
import datetime
from pathlib import Path
from typing import Callable, Any


def get_default_runs_dir() -> Path:
    """Restituisce il percorso della cartella principale 'runs/'."""
    # Cerchiamo la cartella 'runs/' a livello di root del laboratorio
    runs_dir = Path("runs").resolve()
    runs_dir.mkdir(parents=True, exist_ok=True)
    (runs_dir / "favorites").mkdir(parents=True, exist_ok=True)
    (runs_dir / "successi").mkdir(parents=True, exist_ok=True)
    return runs_dir


def get_default_success_dir() -> Path:
    """Restituisce il percorso della cartella 'runs/successi/' per le run con esito >= 80%."""
    succ_dir = get_default_runs_dir() / "successi"
    succ_dir.mkdir(parents=True, exist_ok=True)
    idx_file = succ_dir / "index.json"
    if not idx_file.exists():
        idx_file.write_text("[]\n", encoding="utf-8")
    return succ_dir


def create_run_directory(
    mission: str,
    base_dir: Path | str | None = None,
    custom_name: str | None = None
) -> Path:
    """
    Crea una nuova directory timestampata per la sessione corrente in 'runs/'.
    Esempio: runs/run_20261114_154500_luna_standard/
    """
    base = Path(base_dir).resolve() if base_dir else get_default_runs_dir()
    base.mkdir(parents=True, exist_ok=True)

    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    if custom_name:
        clean_name = "".join(c if c.isalnum() or c in ("-", "_") else "_" for c in custom_name)
        dir_name = f"run_{timestamp}_{clean_name}"
    else:
        dir_name = f"run_{timestamp}_{mission}"

    run_dir = base / dir_name
    run_dir.mkdir(parents=True, exist_ok=True)
    (run_dir / "videos").mkdir(parents=True, exist_ok=True)
    return run_dir


def extract_reward_function_code(reward_fn: Callable) -> str:
    """Estrae in modo sicuro il codice sorgente della funzione scritta dallo studente."""
    try:
        source = inspect.getsource(reward_fn)
        return source.strip()
    except Exception:
        return "# Codice sorgente non disponibile tramite inspect."


def save_run_metadata(
    run_dir: Path,
    mission: str,
    mission_cfg: dict[str, Any],
    env_kwargs: dict[str, Any],
    total_timesteps: int,
    elapsed_seconds: float,
    reward_fn: Callable,
    eval_stats: dict[str, Any],
    diagnostics: dict[str, Any],
    tag: str = "Standard",
    notes: str = "",
) -> Path:
    """
    Salva il file JSON con tutti i metadati e il file Python con il codice sorgente esatto
    della custom_reward utilizzata per questa run.
    """
    source_code = extract_reward_function_code(reward_fn)

    # 1. Salva il codice Python autonomo per consultazione rapida
    code_path = run_dir / "reward_function.py"
    try:
        code_path.write_text(
            f'"""\nFunzione di Reward registrata per la run: {run_dir.name}\nData: {datetime.datetime.now().isoformat()}\n"""\n\n'
            f"{source_code}\n",
            encoding="utf-8"
        )
    except Exception:
        pass

    metadata = {
        "run_id": run_dir.name,
        "timestamp": datetime.datetime.now().isoformat(),
        "tag": tag,
        "notes": notes,
        "mission": {
            "key": mission,
            "title": mission_cfg.get("title", mission),
            "description": mission_cfg.get("description", ""),
            "goal": mission_cfg.get("goal", ""),
            "env_kwargs": env_kwargs,
        },
        "training": {
            "total_timesteps": total_timesteps,
            "elapsed_seconds": round(elapsed_seconds, 2),
            "fps_average": int(total_timesteps / max(1.0, elapsed_seconds)),
        },
        "eval_metrics": {
            "eval_success_rate": eval_stats.get("eval_success_rate", 0.0),
            "eval_successes": eval_stats.get("eval_successes", 0),
            "eval_crashes": eval_stats.get("eval_crashes", 0),
            "eval_timeouts": eval_stats.get("eval_timeouts", 0),
            "total_evaluated_episodes": eval_stats.get("total_evaluated_episodes", 0),
            "mean_reward": eval_stats.get("mean_reward", 0.0),
            "std_reward": eval_stats.get("std_reward", 0.0),
            "mean_length": eval_stats.get("mean_length", 0.0),
            "mean_fuel": eval_stats.get("mean_fuel", 0.0),
            "mean_fuel_success": eval_stats.get("mean_fuel_success", 0.0),
            "mean_fuel_failure": eval_stats.get("mean_fuel_failure", 0.0),
        },
        "diagnostics": {
            "status": diagnostics.get("status", "unknown"),
            "title": diagnostics.get("title", ""),
            "suggestion": diagnostics.get("suggestion", ""),
        },
        "reward_source_code": source_code,
    }

    meta_file = run_dir / "run_metadata.json"
    meta_file.write_text(json.dumps(metadata, indent=2, ensure_ascii=False), encoding="utf-8")
    return meta_file


def list_all_runs(runs_dir: Path | str | None = None) -> list[dict[str, Any]]:
    """Restituisce l'elenco di tutte le run salvate con i relativi metadati sintetici."""
    base = Path(runs_dir).resolve() if runs_dir else get_default_runs_dir()
    if not base.exists():
        return []

    runs = []
    for sub in sorted(base.iterdir(), reverse=True):
        if not sub.is_dir() or sub.name in ("favorites", "successi"):
            continue
        meta_file = sub / "run_metadata.json"
        if meta_file.exists():
            try:
                data = json.loads(meta_file.read_text(encoding="utf-8"))
                data["path"] = str(sub)
                runs.append(data)
            except Exception:
                continue
    return runs


def archivia_esperimento_successo(
    run_dir: Path | str,
    nome_run: str = "successo",
    succ_rate: float = 80.0,
    eval_stats: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """
    Archivia automaticamente un esperimento con tasso di successo >= 80%
    nella cartella dedicata 'runs/successi/', separata dai preferiti dell'utente.
    Copia checkpoint, video, grafici, codice sorgente e genera la Scheda Tecnica PDF.
    """
    import shutil
    run_dir = Path(run_dir)
    eval_stats = eval_stats or {}
    succ_base = get_default_success_dir()

    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    clean_name = "".join(c if c.isalnum() or c in ("-", "_") else "_" for c in nome_run)
    dest_name = f"successo_{clean_name}_{timestamp}"
    dest_dir = succ_base / dest_name
    dest_dir.mkdir(parents=True, exist_ok=True)

    # Copia i file generati dalla run
    for item in run_dir.iterdir():
        if item.is_file():
            shutil.copy2(item, dest_dir / item.name)
        elif item.is_dir() and item.name == "videos":
            shutil.copytree(item, dest_dir / "videos", dirs_exist_ok=True)

    # Genera la Scheda Tecnica nella cartella di destinazione
    from play2learn_lunarlander.reporting import genera_report_esperimento
    rep_files = genera_report_esperimento(
        run_or_fav=dest_dir,
        output_dir=dest_dir,
    )

    # Aggiorna l'indice dei successi
    idx_file = succ_base / "index.json"
    index_data = []
    if idx_file.exists():
        try:
            index_data = json.loads(idx_file.read_text(encoding="utf-8"))
        except Exception:
            index_data = []

    entry = {
        "id": dest_name,
        "name": nome_run,
        "success_rate": round(succ_rate, 1),
        "mean_fuel": round(float(eval_stats.get("mean_fuel", 0.0)), 1),
        "mean_length": round(float(eval_stats.get("mean_length", 0.0)), 1),
        "archived_at": datetime.datetime.now().strftime("%d/%m/%Y %H:%M"),
        "dir_path": str(dest_dir),
        "pdf_report": str(rep_files.get("pdf_path", "")),
        "video_count": len(list((dest_dir / "videos").glob("*.mp4"))) if (dest_dir / "videos").exists() else 0,
    }
    index_data.insert(0, entry)
    idx_file.write_text(json.dumps(index_data, indent=2, ensure_ascii=False), encoding="utf-8")

    return {
        "dest_dir": dest_dir,
        "rep_files": rep_files,
        "entry": entry,
    }

def find_run_by_timestamp(
    timestamp_or_id: str | Path | None = None,
    runs_dir: Path | str | None = None,
) -> Path | None:
    """
    Individua la cartella di una run a partire dal suo timestamp (es. '20261115_210530'),
    da una frazione di esso o dal nome della cartella (anche di più esperimenti fa).
    Se il parametro è vuoto o 'latest', restituisce la run più recente.
    """
    base = Path(runs_dir).resolve() if runs_dir else get_default_runs_dir()
    if not base.exists():
        return None

    # Se è già un Path valido su disco
    if timestamp_or_id is not None:
        p = Path(timestamp_or_id)
        if p.exists() and p.is_dir() and (p / "run_metadata.json").exists():
            return p.resolve()
        if (base / p).exists() and (base / p).is_dir() and ((base / p) / "run_metadata.json").exists():
            return (base / p).resolve()
        if (base / "successi" / p).exists() and ((base / "successi" / p) / "run_metadata.json").exists():
            return (base / "successi" / p).resolve()

    query = str(timestamp_or_id or "").strip()
    is_latest = query.lower() in ("", "latest", "ultima", "ultimo", "now")
    is_previous = query.lower() in ("previous", "penultima", "penultimo", "second_latest", "seconda_ultima", "prev")

    # Raccoglie tutte le directory candidate da runs/, runs/successi/ e runs/favorites/
    all_dirs: list[Path] = []

    # 1. runs standard dirette
    for sub in base.iterdir():
        if sub.is_dir() and sub.name not in ("favorites", "successi"):
            if (sub / "run_metadata.json").exists():
                all_dirs.append(sub)

    # 2. runs in successi/
    succ_dir = base / "successi"
    if succ_dir.exists():
        for sub in succ_dir.iterdir():
            if sub.is_dir() and (sub / "run_metadata.json").exists():
                all_dirs.append(sub)

    # 3. runs in favorites/
    fav_dir = base / "favorites"
    if fav_dir.exists():
        for sub in fav_dir.iterdir():
            if sub.is_dir() and (sub / "run_metadata.json").exists():
                all_dirs.append(sub)

    if not all_dirs:
        return None

    # Ordina per data di modifica più recente
    all_dirs.sort(key=lambda d: d.stat().st_mtime, reverse=True)

    if is_latest:
        return all_dirs[0]
    if is_previous:
        return all_dirs[1] if len(all_dirs) > 1 else all_dirs[0]

    # Pulizia query: rimozione caratteri non alfanumerici per confronto flessibile
    clean_query = query.lower().replace("-", "").replace(":", "").replace(" ", "_")

    # Ricerca esatta o parziale
    for d in all_dirs:
        d_name = d.name.lower()
        if clean_query in d_name or query.lower() in d_name:
            return d

        # Controlla nei metadati se il timestamp o run_id corrispondono
        meta_file = d / "run_metadata.json"
        if meta_file.exists():
            try:
                meta = json.loads(meta_file.read_text(encoding="utf-8"))
                ts_meta = meta.get("timestamp", "").lower().replace("-", "").replace(":", "")
                run_id = meta.get("run_id", "").lower()
                if clean_query in ts_meta or clean_query in run_id or query.lower() in run_id:
                    return d
            except Exception:
                continue

    return None


def elenca_cronologia_run(runs_dir: Path | str | None = None) -> list[dict[str, Any]]:
    """
    Restituisce l'elenco cronologico di tutte le run salvate (standard, successi e preferiti),
    estraendo per ciascuna il timestamp esatto, il nome, l'esito percentuale e il consumo.
    """
    import re
    base = Path(runs_dir).resolve() if runs_dir else get_default_runs_dir()
    if not base.exists():
        return []

    results: list[dict[str, Any]] = []

    def _process_dir(d: Path, category: str) -> None:
        meta_file = d / "run_metadata.json"
        if not meta_file.exists():
            return
        try:
            meta = json.loads(meta_file.read_text(encoding="utf-8"))
        except Exception:
            return

        ts_match = re.search(r"(\d{8}_\d{6})", d.name)
        if ts_match:
            ts_str = ts_match.group(1)
        else:
            raw_ts = meta.get("timestamp", "")
            ts_str = raw_ts.replace("-", "")[:8] + "_" + raw_ts.replace(":", "")[9:15] if len(raw_ts) >= 19 else d.name

        metrics = meta.get("eval_metrics", {})
        mission = meta.get("mission", {})
        succ_rate = float(metrics.get("eval_success_rate", 0.0))

        results.append({
            "timestamp": ts_str,
            "run_id": d.name,
            "name": meta.get("tag") or mission.get("title") or d.name,
            "mission": mission.get("key", ""),
            "success_rate": succ_rate,
            "mean_fuel": float(metrics.get("mean_fuel", 0.0)),
            "mean_length": float(metrics.get("mean_length", 0.0)),
            "category": category,
            "dir_path": str(d),
            "mtime": d.stat().st_mtime,
            "has_model": (d / "best_model.zip").exists() or (d / "ppo_lunar_lander.zip").exists(),
            "video_count": len(list((d / "videos").glob("*.mp4"))) if (d / "videos").exists() else 0,
        })

    # Standard runs
    for sub in base.iterdir():
        if sub.is_dir() and sub.name not in ("favorites", "successi"):
            _process_dir(sub, "Standard")

    # Successi
    succ_dir = base / "successi"
    if succ_dir.exists():
        for sub in succ_dir.iterdir():
            if sub.is_dir():
                _process_dir(sub, "Successo (>=80%)")

    # Preferiti
    fav_dir = base / "favorites"
    if fav_dir.exists():
        for sub in fav_dir.iterdir():
            if sub.is_dir():
                _process_dir(sub, "Preferito")

    # Ordina per mtime decrescente (le più recenti in cima)
    results.sort(key=lambda r: r["mtime"], reverse=True)
    return results

"""

def salva_nei_preferiti(
    run_or_result: Any,
    nome_personalizzato: str | None = None,
    note: str = "",
    tag: str = "Mio Preferito",
) -> dict[str, Any]:
    """
    Salva una run o un video nell'archivio dei Preferiti Personali dello studente.
    Copia in modo sicuro modello, grafici, video e codice in 'runs/favorites/'.

    Parametri:
    - run_or_result: Risultato restituito da `run_lab(...)`, percorso cartella run, o percorso a un file .mp4.
    - nome_personalizzato: Nome descrittivo (es. "Allunaggio_Stabile_v2").
    - note: Breve commento sulle scelte matematiche effettuate.
    - tag: Categoria (default: "Mio Preferito").
    """
    # Se è un file video .mp4, delega a salva_video_preferito
    if isinstance(run_or_result, (str, Path)) and str(run_or_result).lower().endswith(".mp4"):
        return salva_video_preferito(run_or_result, nome=nome_personalizzato, note=note, tag=tag)

    fav_dir = _get_favorites_dir()

    # 1. Risoluzione della cartella sorgente della run
    source_dir: Path | None = None
    if isinstance(run_or_result, dict):
        if "checkpoint_dir" in run_or_result:
            source_dir = Path(run_or_result["checkpoint_dir"])
        elif "path" in run_or_result:
            source_dir = Path(run_or_result["path"])
        elif "run_dir" in run_or_result:
            source_dir = Path(run_or_result["run_dir"])
    elif isinstance(run_or_result, (str, Path)):
        source_dir = Path(run_or_result)

    if not source_dir or not source_dir.exists():
        raise ValueError(f"[Play2Learn] Impossibile trovare la cartella della run: {run_or_result}")

    # 2. Lettura metadati esistenti
    meta_path = source_dir / "run_metadata.json"
    if meta_path.exists():
        metadata = json.loads(meta_path.read_text(encoding="utf-8"))
    else:
        metadata = {
            "run_id": source_dir.name,
            "mission": {"title": "Missione sconosciuta", "key": "luna_standard"},
            "eval_metrics": {},
        }

    # 3. Creazione destinazione nel folder favorites
    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    clean_title = nome_personalizzato or metadata.get("mission", {}).get("key", "modello")
    clean_title = "".join(c if c.isalnum() or c in ("-", "_") else "_" for c in clean_title)
    dest_name = f"fav_{clean_title}_{timestamp}"
    dest_dir = fav_dir / dest_name
    dest_dir.mkdir(parents=True, exist_ok=True)

    # 4. Copia dei file essenziali (modello, grafici, codice, video/gif)
    for item in source_dir.iterdir():
        if item.is_file():
            shutil.copy2(item, dest_dir / item.name)
        elif item.is_dir() and item.name == "videos":
            shutil.copytree(item, dest_dir / "videos", dirs_exist_ok=True)

    # 5. Aggiornamento metadati preferito
    metadata["favorite_info"] = {
        "display_name": nome_personalizzato or dest_name,
        "saved_at": datetime.datetime.now().isoformat(),
        "tag": tag,
        "notes": note,
        "favorite_dir": str(dest_dir),
    }
    (dest_dir / "run_metadata.json").write_text(
        json.dumps(metadata, indent=2, ensure_ascii=False),
        encoding="utf-8"
    )

    # 6. Aggiornamento indice globale preferiti
    index_data = _load_favorites_index()
    entry = {
        "id": dest_name,
        "type": "run",
        "name": nome_personalizzato or dest_name,
        "tag": tag,
        "notes": note,
        "saved_at": datetime.datetime.now().strftime("%d/%m/%Y %H:%M"),
        "mission_title": metadata.get("mission", {}).get("title", "Lunar Lander"),
        "success_rate": metadata.get("eval_metrics", {}).get("eval_success_rate", 0.0),
        "mean_fuel": metadata.get("eval_metrics", {}).get("mean_fuel", 0.0),
        "mean_length": metadata.get("eval_metrics", {}).get("mean_length", 0.0),
        "dir_path": str(dest_dir),
    }
    index_data.insert(0, entry)
    _save_favorites_index(index_data)

    print("\n" + "=" * 70)
    print(f"  ⭐ MODELLO SALVATO NEI PREFERITI PERSONALI!")
    print(f"  🏷️  Nome: {entry['name']}")
    print(f"  🏆 Allunaggi Riusciti: {entry['success_rate']:.0f}% | Carburante Medio: {entry['mean_fuel']:.1f}")
    if note:
        print(f"  📝 Note: {note}")
    print(f"  📁 Cartella: runs/favorites/{dest_dir.name}")
    print("=" * 70 + "\n")

    return entry


def salva_video_preferito(
    video_ref: Any,
    nome: str | None = None,
    note: str = "",
    tag: str = "Video Preferito",
) -> dict[str, Any]:
    """
    Salva un singolo video tra i preferiti personali dell'utente.

    Parametri:
    - video_ref: percorso a un file .mp4, oppure oggetto LabResult (prende il primo video di successo).
    - nome: titolo descrittivo per il video (es. 'Mio Allunaggio Spettacolare').
    - note: annotazione didattica dello studente.
    - tag: categoria/etichetta (default: 'Video Preferito').
    """
    fav_dir = _get_favorites_dir()
    video_store = fav_dir / "videos"
    video_store.mkdir(parents=True, exist_ok=True)

    v_path: Path | None = None
    if isinstance(video_ref, (str, Path)):
        p = Path(video_ref)
        if p.exists() and p.suffix.lower() == ".mp4":
            v_path = p
    elif isinstance(video_ref, dict):
        succ_vids = video_ref.get("success_videos", []) or video_ref.get("recorded_videos", [])
        if succ_vids and Path(succ_vids[0]).exists():
            v_path = Path(succ_vids[0])

    if not v_path or not v_path.exists():
        raise ValueError(f"[Play2Learn] Video non trovato o non valido: {video_ref}")

    timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
    clean_name = nome or v_path.stem
    clean_name = "".join(c if c.isalnum() or c in ("-", "_") else "_" for c in clean_name)
    dest_video_name = f"fav_vid_{clean_name}_{timestamp}.mp4"
    dest_video_path = video_store / dest_video_name
    shutil.copy2(v_path, dest_video_path)

    # Aggiorna l'indice globale dei preferiti
    index_data = _load_favorites_index()
    entry = {
        "id": dest_video_name,
        "type": "video",
        "name": nome or clean_name,
        "tag": tag,
        "notes": note,
        "saved_at": datetime.datetime.now().strftime("%d/%m/%Y %H:%M"),
        "mission_title": "Video Allunaggio",
        "video_path": str(dest_video_path),

def carica_preferito(identifier: str | int = 0) -> tuple[PPO, dict[str, Any]]:
    """
    Ricarica il modello PPO e i relativi metadati da un elemento salvato nei Preferiti.
    Può essere specificato l'indice numerico (0 = il più recente) o il nome.
    Ideale per una dimostrazione dal vivo durante il colloquio orale!
    """
    entries = _load_favorites_index()
    if not entries:
        raise ValueError("[Play2Learn] La lista dei preferiti è vuota.")

    selected_entry: dict[str, Any] | None = None
    if isinstance(identifier, int):
        if 0 <= identifier < len(entries):
            selected_entry = entries[identifier]
        else:
            raise IndexError(f"Indice preferito non valido: {identifier} (disponibili 0..{len(entries)-1})")
    else:
        for e in entries:
            if e.get("id") == identifier or e.get("name") == identifier:
                selected_entry = e
                break

    if not selected_entry:
        raise ValueError(f"Nessun preferito trovato con identificativo: {identifier}")

    if selected_entry.get("type") == "video":
        raise ValueError(
            f"[Play2Learn] L'elemento '{selected_entry['name']}' è un video salvato nei preferiti, non una rete neurale.\n"
            f"File video: {selected_entry.get('video_path')}"
        )

    dest_dir = Path(selected_entry["dir_path"])
    # Cerca il file modello
    model_path = dest_dir / "best_model.zip"
    if not model_path.exists():
        model_path = dest_dir / "ppo_lunar_lander.zip"
    if not model_path.exists():
        zip_files = list(dest_dir.glob("*.zip"))
        if zip_files:
            model_path = zip_files[0]
        else:
            raise FileNotFoundError(f"Nessun checkpoint modello .zip trovato in: {dest_dir}")

    try:
        from stable_baselines3 import PPO
    except ImportError as e:
        raise ImportError(
            "❌ stable-baselines3 non trovato! Assicurati di aver eseguito la Cella di installazione:\n"
            "!pip install -q stable-baselines3"
        ) from e

    model = PPO.load(str(model_path))
    meta_path = dest_dir / "run_metadata.json"
    meta = json.loads(meta_path.read_text(encoding="utf-8")) if meta_path.exists() else selected_entry

    print(f"✅ Modello '{selected_entry['name']}' caricato con successo da: {dest_dir.name}")
    return model, meta

def genera_pdf_report(

    Genera una scheda tecnica PDF vettoriale (2 pagine A4 ad alta risoluzione 300 DPI)
    con telemetria di volo, confronto tra successi e fallimenti, codice sorgente
    integrale della custom reward e note di debug.
    """
    pdf_path = Path(pdf_path)
    pdf_path.parent.mkdir(parents=True, exist_ok=True)

    mission = meta.get("mission", {})
    mission_title = mission.get("title", "Lunar Lander")
    env_kwargs = mission.get("env_kwargs", {})

    fuel_succ = metrics.get("mean_fuel_success", fuel_used)
    fuel_fail = metrics.get("mean_fuel_failure", 0.0)
    succ_count = metrics.get("eval_successes", 0)
    fail_count = metrics.get("eval_crashes", 0) + metrics.get("eval_timeouts", 0)
    flight_steps = metrics.get("mean_length", 0.0)
    code_source = meta.get("reward_source_code", "# Codice reward non disponibile.")

    if not note_studente:
        note_studente = meta.get("notes") or meta.get("favorite_info", {}).get("notes") or "Nessuna annotazione inserita."

    with PdfPages(str(pdf_path)) as pdf:
        # =========================================================================
        # PAGINA 1: Intestazione, Tabella Telemetrica e Grafico di Apprendimento 2x2
        # =========================================================================
        fig1 = Figure(figsize=(8.27, 11.69), dpi=300)
        FigureCanvasAgg(fig1)

        # Header
        fig1.text(0.5, 0.955, "SCHEDA TECNICA ESPERIMENTO — REINFORCEMENT LEARNING",
                  ha="center", va="top", fontsize=13.5, fontweight="bold", color="#1a237e")
        fig1.text(0.5, 0.932, f"{titolo_esperimento}   |   Autore: {autore}   |   Data: {datetime.datetime.now().strftime('%d/%m/%Y %H:%M')}",
                  ha="center", va="top", fontsize=10.0, color="#333333")
        fig1.text(0.5, 0.912, f"Missione: {mission_title}  (Gravità: {env_kwargs.get('gravity', -10.0)}, Vento: {env_kwargs.get('enable_wind', False)})",
                  ha="center", va="top", fontsize=9.2, style="italic", color="#555555")

        # Linea orizzontale
        ax_line = fig1.add_axes([0.06, 0.898, 0.88, 0.002])
        ax_line.axhline(0, color="#1a237e", lw=2)
        ax_line.axis("off")

        # 1. Tabella Telemetrica Valutazione
        fig1.text(0.06, 0.880, "1. Risultati della Valutazione (Analisi Successi e Fallimenti)",
                  fontsize=11.0, fontweight="bold", color="#283593")

        col_labels = ["Metrica Telemetrica", "Valore Sperimentale", "Riferimento Ideale", "Stato"]
        fuel_fail_val = f"{fuel_fail:.1f} unità" if fail_count > 0 else "0 fallimenti"
        fuel_fail_ref = "Nessun fallimento (0 un.)"
        fuel_fail_stato = "OTTIMO" if fail_count == 0 else "DIAGNOSI"

        table_vals = [
            ["Tasso di Allunaggio Riuscito", f"{success_rate:.1f}%", ">= 80.0%", "OBIETTIVO RAGGIUNTO" if success_rate >= 80 else "IN OTTIMIZZAZIONE"],
            ["Prove di Valutazione", f"{succ_count} successi / {fail_count} fallimenti", "Campione misto", "CAMPIONATO"],
            ["Carburante Medio Voli Riusciti", f"{fuel_succ:.1f} unità", "< 60.0 unità", "OTTIMO" if fuel_succ <= 60 else "STANDARD"],
            ["Carburante Medio Voli Falliti", fuel_fail_val, fuel_fail_ref, fuel_fail_stato],
            ["Durata Media Volo", f"{flight_steps:.0f} step ({flight_steps/50:.1f} s)", "150 - 350 step", "NOMINALE"],
        ]

        ax_table = fig1.add_axes([0.06, 0.705, 0.88, 0.165])
        ax_table.axis("off")
        col_widths = [0.31, 0.24, 0.25, 0.20]
        tab = ax_table.table(
            cellText=table_vals,
            colLabels=col_labels,
            colWidths=col_widths,
            loc="center",
            cellLoc="center",
            colColours=["#e8eaf6"] * 4,
        )
        tab.auto_set_font_size(False)
        tab.set_fontsize(8.5)
        tab.scale(1.0, 1.45)

        for (row, col), cell in tab.get_celld().items():
            cell.set_edgecolor("#c5cae9")
            if row == 0:
                cell.set_text_props(weight="bold", color="#1a237e")
            if col == 0 and row > 0:
                cell.set_text_props(horizontalalignment="left")
            else:
                cell.set_text_props(horizontalalignment="center")

        # 2. Inserimento Immagine Grafico 2x2 ad alta definizione
        fig1.text(0.06, 0.675, "2. Curve di Apprendimento e Telemetria di Volo",
                  fontsize=11.0, fontweight="bold", color="#283593")

        if plot_path and Path(plot_path).exists():
            img = plt.imread(str(plot_path))
            ax_img = fig1.add_axes([0.06, 0.04, 0.88, 0.62])
            ax_img.imshow(img, interpolation="antialiased")
            ax_img.axis("off")

        pdf.savefig(fig1)
        fig1.clear()

        # =========================================================================
        # PAGINA 2: Codice Sorgente Integrale della Custom Reward e Note di Debug
        # =========================================================================
        fig2 = Figure(figsize=(8.27, 11.69), dpi=300)
        FigureCanvasAgg(fig2)

        fig2.text(0.5, 0.955, "FORMULAZIONE MATEMATICA E NOTE DI DEBUG",
                  ha="center", va="top", fontsize=13.5, fontweight="bold", color="#1a237e")

        fig2.text(0.06, 0.915, "3. Codice Sorgente Completo della Funzione di Reward (Reward Shaping):",
                  fontsize=11.0, fontweight="bold", color="#283593")

        # Pulizia righe e calcolo altezza
        clean_code = "\n".join(line.rstrip() for line in code_source.strip().splitlines())
        code_lines = clean_code.splitlines()
        n_lines = len(code_lines)

        # Dimensionamento dinamico del font per visualizzare tutto il codice senza troncamenti
        if n_lines <= 42:
            font_size = 7.8
            line_spacing = 0.0120
        elif n_lines <= 62:
            font_size = 7.1
            line_spacing = 0.0102
        elif n_lines <= 85:
            font_size = 6.4
            line_spacing = 0.0088
        else:
            font_size = 5.8
            line_spacing = 0.0075

        code_height = min(0.64, max(0.20, n_lines * line_spacing + 0.03))
        y_code_top = 0.895
        y_code_bottom = y_code_top - code_height

        fig2.text(
            0.06, y_code_top, clean_code,
            fontsize=font_size, family="monospace", va="top",
            bbox=dict(boxstyle="round,pad=0.6", facecolor="#f8f9fa", edgecolor="#90caf9", lw=1.2)
        )

        # Sezione Note e Osservazioni di Debug dello Studente
        y_notes = max(0.035, y_code_bottom - 0.035)
        fig2.text(0.06, y_notes, "4. Note di Debug e Considerazioni Sperimentali:",
                  fontsize=10.5, fontweight="bold", color="#283593")

        debug_box_content = (
            f"NOTE INSERITE DALL'AUTORE:\n{note_studente}\n\n"
            "SPAZIO PER ANNOTAZIONI MANUALI / MODIFICHE FUTURE:\n"
            "• Quali termini matematici hanno migliorato la stabilità?\n"
            "• Quali comportamenti indesiderati sono stati osservati nei fallimenti?\n"
            "• Prossimi tentativi da sperimentare (pesi, velocità, soglie quota):"
        )

        fig2.text(
            0.06, y_notes - 0.02, debug_box_content,
            fontsize=8.4, va="top", family="sans-serif",
            bbox=dict(boxstyle="round,pad=0.5", facecolor="#fffde7", edgecolor="#fbc02d", lw=1.3)
        )

        pdf.savefig(fig2)
        fig2.clear()

    return pdf_path


def genera_report_esperimento(

    # 1. Risoluzione della directory della run
    source_dir: Path | None = None
    if isinstance(run_or_fav, dict):
        if "checkpoint_dir" in run_or_fav:

        elif "path" in run_or_fav:
            source_dir = Path(run_or_fav["path"])

        elif "run_dir" in run_or_fav:

        p = Path(run_or_fav)
        if p.exists() and p.is_dir():
            source_dir = p
        else:
            from play2learn_lunarlander.storage import find_run_by_timestamp
            source_dir = find_run_by_timestamp(str(run_or_fav))

        raise ValueError(f"[Play2Learn] Nessuna directory di run trovata per: {run_or_fav}")

def confronta_run(
    runs_or_favs: list[Any] | tuple[Any, ...] | str,
    output_image_path: Path | str | None = None
) -> dict[str, Any]:
    """
    Confronta 2 o più run generando un grafico comparativo e una tabella di sintesi a colonne.
    Funziona con:
      - Oggetti LabResult restituite da run_lab(...)
      - Stringhe o liste di timestamp (es. '20261115_210530, 20261115_212000')
      - Parole chiave come 'latest, previous' per confrontare al volo le ultime due run
      - Percorsi a cartelle di run
    """
    from play2learn_lunarlander.storage import find_run_by_timestamp, mostra_cronologia_run

    if not runs_or_favs:
        print("[play2learn] Nessuna run fornita per il confronto.")
        print("💡 Suggerimento: inserisci due timestamp separati da virgola (es. 'latest, previous').")
        return {}

    # Se passata come stringa separata da virgola o punto e virgola
    if isinstance(runs_or_favs, str):
        raw_items = [x.strip() for x in runs_or_favs.replace(";", ",").split(",") if x.strip()]
    elif isinstance(runs_or_favs, (list, tuple)):
        raw_items = list(runs_or_favs)
    else:
        raw_items = [runs_or_favs]

    runs_data = []
    unresolved = []

    for item in raw_items:
        source_dir = None
        if isinstance(item, dict):
            if "checkpoint_dir" in item:
                source_dir = Path(item["checkpoint_dir"])
            elif "dir_path" in item:
                source_dir = Path(item["dir_path"])
            elif "path" in item:
                source_dir = Path(item["path"])
            elif "run_dir" in item:
                source_dir = Path(item["run_dir"])
            else:
                runs_data.append(item)
                continue
        elif isinstance(item, (str, Path)):
            p = Path(item)
            if p.exists() and p.is_dir() and (p / "run_metadata.json").exists():
                source_dir = p
            else:
                # Risoluzione automatica da timestamp o alias (es. "latest", "previous")
                resolved = find_run_by_timestamp(str(item))
                if resolved and resolved.exists():
                    source_dir = resolved
                else:
                    unresolved.append(str(item))

        if source_dir and source_dir.exists():
            meta_path = source_dir / "run_metadata.json"
            if meta_path.exists():
                try:
                    meta = json.loads(meta_path.read_text(encoding="utf-8"))
                    metrics = meta.get("eval_metrics", {})
                    mission = meta.get("mission", {})

                    import re
                    ts_m = re.search(r"(\d{8}_\d{6})", source_dir.name)
                    ts_display = ts_m.group(1) if ts_m else source_dir.name
                    run_name = meta.get("favorite_info", {}).get("display_name") or meta.get("tag") or mission.get("title") or source_dir.name

                    runs_data.append({
                        "name": run_name,
                        "timestamp": ts_display,
                        "display_label": f"{run_name}\n({ts_display})",
                        "eval_success_rate": float(metrics.get("eval_success_rate", 0.0)),
                        "eval_mean_fuel": float(metrics.get("mean_fuel", 0.0)),
                        "eval_mean_length": float(metrics.get("mean_length", 0.0)),
                        "mission": mission.get("title", ""),
                        "dir_name": source_dir.name,
                    })
                except Exception:
                    continue

    if unresolved:
        print(f"⚠️ Attenzione: impossibile trovare le seguenti run dal timestamp inserito: {', '.join(unresolved)}")
        print("💡 Ecco l'elenco delle run disponibili con i relativi timestamp da copiare:")
        mostra_cronologia_run()

    if not runs_data:
        print("[play2learn] Impossibile estrarre i dati per il confronto. Verifica i timestamp forniti.")
        return {}

    save_img = Path(output_image_path) if output_image_path else get_default_runs_dir() / "confronto_run.png"
    plot_comparison(runs_data, save_img)

    # Visualizzazione tabella comparativa arricchita con colonna Timestamp
    try:
        from IPython.display import HTML, Image, display
        rows = []
        for r in runs_data:
            succ = r.get("eval_success_rate", 0.0)
            badge_color = "#2e7d32" if succ >= 80.0 else ("#f57c00" if succ >= 50.0 else "#c62828")
            bg_badge = "#e8f5e9" if succ >= 80.0 else ("#fff3e0" if succ >= 50.0 else "#ffebee")
            rows.append(
                f"<tr>"
                f"<td style='padding:8px 12px; font-family:monospace; font-weight:bold; color:#1a237e; background:#f8f9fa;'>{r.get('timestamp', '')}</td>"
                f"<td style='padding:8px 12px; font-weight:bold;'>{r.get('name')}</td>"
                f"<td style='padding:8px 12px;'>{r.get('mission', '')}</td>"
                f"<td style='padding:8px 12px; text-align:center;'><span style='background:{bg_badge}; color:{badge_color}; padding:3px 8px; border-radius:12px; font-weight:bold;'>{succ:.1f}%</span></td>"
                f"<td style='padding:8px 12px; text-align:center;'>{r.get('eval_mean_fuel', 0):.1f} un.</td>"
                f"<td style='padding:8px 12px; text-align:center;'>{r.get('eval_mean_length', 0):.0f} step</td>"
                f"</tr>"
            )

        html = f"""
        <div style="font-family:'Segoe UI', sans-serif; margin:16px 0;">
            <h3 style="color:#0d47a1; margin-bottom:8px;">📊 Confronto Esperimenti di Apprendimento (Analisi Comparativa)</h3>
            <table style="border-collapse:collapse; width:100%; border:1px solid #ccc; text-align:left; font-size:13.5px;">
                <thead style="background:#e3f2fd; color:#0d47a1;">
                    <tr>
                        <th style="padding:10px 12px;">Timestamp</th>
                        <th style="padding:10px 12px;">Esperimento / Versione</th>
                        <th style="padding:10px 12px;">Missione</th>
                        <th style="padding:10px 12px; text-align:center;">% Allunaggio</th>
                        <th style="padding:10px 12px; text-align:center;">Carburante Medio</th>
                        <th style="padding:10px 12px; text-align:center;">Durata Media</th>
                    </tr>
                </thead>
                <tbody>
                    {"".join(rows)}
                </tbody>
            </table>
        </div>
        """
        display(HTML(html))
        if save_img.exists():
            display(Image(filename=str(save_img)))
    except Exception:
        print("\n" + "=" * 78)
        print("  📊 CONFRONTO ESPERIMENTI (TABELLA COMPARATIVA)")
        print("=" * 78)
        print(f"{'TIMESTAMP':<18} | {'NOME RUN':<24} | {'SUCCESSO':<10} | {'CARBURANTE':<12} | {'DURATA':<10}")
        print("-" * 78)
        for r in runs_data:
            print(f"{r.get('timestamp', ''):<18} | {r.get('name', '')[:23]:<24} | {r.get('eval_success_rate', 0):<9.1f}% | {r.get('eval_mean_fuel', 0):<12.1f} | {r.get('eval_mean_length', 0):<10.0f}")
        print("=" * 78 + "\n")

    return {"runs": runs_data, "comparison_plot": save_img}


def genera_report_da_timestamp(
    timestamp: str = "latest",
    tipo: str = "Base",
    autore: str = "Studente",
    titolo: str | None = None,
    note: str = "",
    n_extra_episodes: int = 35,
    target_success_videos: int = 4,
    target_failure_videos: int = 4,
    output_dir: Path | str | None = None,
) -> dict[str, Any]:
    """
    Genera la reportistica tecnica completa (Scheda Tecnica PDF 300 DPI, HTML, Markdown)
    a partire dal timestamp (es. '20261115_210530') o dal nome di una run passata.

    Parametri:
      - timestamp: stringa del timestamp (es. '20261115_210530') oppure 'latest' per la run più recente.
      - tipo: 'Base' (genera il report con i dati archiviati) oppure
              'Avanzata' (ricarica il modello, esegue nuove run di verifica e registra più video di test).
      - autore: nome dello studente o gruppo.
      - titolo: titolo personalizzato dell'esperimento (opzionale).
      - note: note personali e considerazioni di debug.
      - n_extra_episodes: numero di episodi di sweep per la valutazione avanzata (default 35).
      - target_success_videos: video di successo da registrare in modalità avanzata (default 4).
      - target_failure_videos: video di fallimento da registrare in modalità avanzata (default 4).
      - output_dir: cartella di salvataggio (opzionale, default sottocartella timestampata della run).
    """
    from play2learn_lunarlander.storage import find_run_by_timestamp, mostra_cronologia_run
    from play2learn_lunarlander.visualizer import display_notebook_summary

    print("\n" + "=" * 78)
    print("  🕒 GENERATORE DI REPORT DA TIMESTAMP — LUNAR LANDER")
    print("=" * 78)

    # 1. Ricerca della directory della run dal timestamp
    run_dir = find_run_by_timestamp(timestamp)
    if run_dir is None:
        print(f"❌ Nessuna run trovata corrispondente a: '{timestamp}'")
        print("💡 Ecco l'elenco delle run disponibili con i relativi timestamp da copiare:\n")
        mostra_cronologia_run()
        return {}

    print(f"🔍 Run individuata con successo: runs/{run_dir.name}")

    meta_file = run_dir / "run_metadata.json"
    if meta_file.exists():
        try:
            meta = json.loads(meta_file.read_text(encoding="utf-8"))
        except Exception:
            meta = {}
    else:
        meta = {}

    mission = meta.get("mission", {})
    mission_title = mission.get("title", "Lunar Lander")
    env_kwargs = mission.get("env_kwargs", {})

    if not titolo:
        titolo = f"Esperimento {mission_title} — Reinforcement Learning"
    if not note:
        note = meta.get("notes") or meta.get("favorite_info", {}).get("notes") or ""

    is_avanzata = "avanzat" in str(tipo).lower()

    # 2. Se modalità Avanzata: esegui nuove run di test e registra più video
    if is_avanzata:
        print("\n🚀 [Modalità AVANZATA attiva] Avvio sessione di verifica estesa per registrare e salvare più video...")
        model_path = None
        for cand in ("best_model.zip", "ppo_lunar_lander.zip"):
            if (run_dir / cand).exists():
                model_path = run_dir / cand
                break
        if model_path is None:
            zips = list(run_dir.glob("*.zip"))
            if zips:
                model_path = zips[0]

        if model_path is not None:
            try:
                from stable_baselines3 import PPO
                from play2learn_lunarlander.evaluator import evaluate_and_record
                from play2learn_lunarlander.config import ENV_ID

                print(f"📦 Caricamento modello salvato da: {model_path.name}")
                model = PPO.load(str(model_path))

                # Ricostruzione della funzione custom_reward dalla run
                reward_fn = None
                reward_file = run_dir / "reward_function.py"
                if reward_file.exists():
                    try:
                        code_txt = reward_file.read_text(encoding="utf-8")
                        scope = {}
                        exec(code_txt, scope)
                        reward_fn = scope.get("custom_reward")
                    except Exception:
                        reward_fn = None

                if reward_fn is None and "reward_source_code" in meta:
                    try:
                        scope = {}
                        exec(meta["reward_source_code"], scope)
                        reward_fn = scope.get("custom_reward")
                    except Exception:
                        reward_fn = None

                if reward_fn is None:
                    reward_fn = lambda obs, act, s, c, t: 0.0

                env_id = meta.get("env_id") or ENV_ID
                videos_dir = run_dir / "videos"
                videos_dir.mkdir(parents=True, exist_ok=True)

                print(f"🎬 Registrazione di nuovi video di test in: runs/{run_dir.name}/videos/")
                adv_stats = evaluate_and_record(
                    model=model,
                    env_id=env_id,
                    reward_fn=reward_fn,
                    video_dir=videos_dir,
                    target_successes=target_success_videos,
                    target_failures=target_failure_videos,
                    sweep_timeout_seconds=8.0,
                    min_eval_episodes=n_extra_episodes,
                    max_eval_episodes=n_extra_episodes + 30,
                    seed=int(datetime.datetime.now().timestamp()) % 10000,
                    env_kwargs=env_kwargs,
                    create_gif=False,
                    filename_prefix="avanzato",
                )

                meta["eval_metrics_avanzata"] = adv_stats
                meta["eval_metrics"] = adv_stats
                meta_file.write_text(json.dumps(meta, indent=2, ensure_ascii=False), encoding="utf-8")
                print(f"✅ Sessione avanzata completata: {adv_stats.get('eval_success_rate', 0.0):.1f}% di successo su {adv_stats.get('total_evaluated_episodes', 0)} voli!")
            except Exception as e:
                print(f"⚠️ Impossibile completare la sessione estesa ({e}). Utilizzo dati pregressi.")
        else:
            print(f"ℹ️ File pesi del modello (.zip) non trovato in '{run_dir.name}'. Verranno utilizzati i video e i dati esistenti.")
    else:
        print("📄 [Modalità Base] Generazione report dai dati e video archiviati per la run.")

    # 3. Generazione report (PDF 300 DPI, HTML, MD)
    rep_files = genera_report_esperimento(
        run_or_fav=run_dir,
        autore=autore,
        titolo=titolo,
        note=note,
        output_dir=output_dir,
    )

    # 4. Visualizzazione inline dei video e del grafico nel notebook
    plot_path = run_dir / "reward_plot.png"
    vids_dir = run_dir / "videos"
    all_videos = sorted(vids_dir.glob("*.mp4"), key=lambda p: p.stat().st_mtime, reverse=True) if vids_dir.exists() else []

    print("\n" + "=" * 78)
    print("  📊 TELEMETRIA DI VOLO E VIDEO SALVATI:")
    print("=" * 78)
    display_notebook_summary(
        plot_path=plot_path if plot_path.exists() else "",
        videos=all_videos,
        max_videos=6 if is_avanzata else 3,
    )

    print("\n" + "=" * 78)
    print(f"  ✨ REPORT ({tipo.upper()}) PRONTO E SALVATO CON SUCCESSO!")
    print(f"  • Scheda Tecnica PDF (300 DPI): {rep_files.get('pdf_path')}")
    print(f"  • Report HTML Interattivo:     {rep_files.get('html_path')}")
    print(f"  • Documento Markdown:          {rep_files.get('md_path')}")
    print("=" * 78 + "\n")

    return {
        "run_dir": run_dir,
        "meta": meta,
        "rep_files": rep_files,
        "videos": all_videos,
    }

"""
Script di Reset per il Laboratorio di Reinforcement Learning (Lunar Lander).
Posizionato all'interno di play2learn_lunarlander/ per mantenere la root pulita.

Ripristina l'ambiente allo stato originale e pulito:
1. Svuota la cartella 'runs/' (eliminando tutte le prove, video, modelli e report di test).
2. Ricrea la struttura vergine 'runs/favorites/index.json'.
3. Azzera tutti gli output e contatori di esecuzione nel Notebook (.ipynb).
4. Rimuove la cache Python (__pycache__ e file .pyc).
"""

from __future__ import annotations

import os
import sys
import json
import shutil
from pathlib import Path


def get_root_dir() -> Path:
    """Restituisce la root directory del laboratorio (cartella genitore del package)."""
    current_dir = Path(__file__).resolve().parent
    # Se il file si trova dentro play2learn_lunarlander/, la root è il genitore
    if current_dir.name == "play2learn_lunarlander":
        return current_dir.parent
    return current_dir


def pulisci_notebook(nb_path: Path) -> int:
    """Azzera tutti gli output e execution_count delle celle del notebook."""
    if not nb_path.exists():
        print(f"⚠️  Notebook non trovato in: {nb_path.name}")
        return 0

    try:
        nb = json.loads(nb_path.read_text(encoding="utf-8"))
        cleared_count = 0
        for cell in nb.get("cells", []):
            if cell.get("cell_type") == "code":
                if cell.get("outputs") or cell.get("execution_count") is not None:
                    cell["outputs"] = []
                    cell["execution_count"] = None
                    cleared_count += 1

        nb_path.write_text(json.dumps(nb, indent=1, ensure_ascii=False), encoding="utf-8")
        size_kb = nb_path.stat().st_size / 1024
        print(f"✅ Notebook ripulito: {nb_path.name} ({size_kb:.1f} KB, azzerate {cleared_count} celle di output)")
        return cleared_count
    except Exception as e:
        print(f"❌ Errore durante la pulizia del notebook: {e}")
        return 0


def pulisci_runs(runs_dir: Path) -> None:
    """Elimina tutte le run precedenti e ricrea l'alberatura runs/favorites vergine."""
    # Controllo di sicurezza: verifica che il target sia effettivamente la cartella 'runs'
    if runs_dir.name != "runs":
        raise ValueError(f"❌ Errore di sicurezza: la cartella target non è 'runs' ma '{runs_dir.name}'. Operazione bloccata!")

    if runs_dir.exists():
        for item in runs_dir.iterdir():
            if item.is_dir():
                shutil.rmtree(item, ignore_errors=True)
            elif item.is_file():
                item.unlink(missing_ok=True)
    else:
        runs_dir.mkdir(parents=True, exist_ok=True)

    # Ricrea la struttura minima pulita
    fav_dir = runs_dir / "favorites"
    fav_dir.mkdir(parents=True, exist_ok=True)
    (fav_dir / "index.json").write_text("[]\n", encoding="utf-8")
    (fav_dir / "videos").mkdir(parents=True, exist_ok=True)

    succ_dir = runs_dir / "successi"

def pulisci_cache(root_dir: Path) -> None:
    """Rimuove unicamente le cartelle __pycache__, i file .pyc e i checkpoint di Jupyter."""
    rimossi = 0
    for p in root_dir.rglob("__pycache__"):
        if p.is_dir():
            shutil.rmtree(p, ignore_errors=True)
            rimossi += 1
    for p in root_dir.rglob(".ipynb_checkpoints"):
        if p.is_dir():
            shutil.rmtree(p, ignore_errors=True)
            rimossi += 1
    for p in root_dir.rglob("*.pyc"):
        if p.is_file():
            p.unlink(missing_ok=True)
            rimossi += 1
    if rimossi > 0:
        print(f"✅ Cache Python e checkpoint temporanei rimossi ({rimossi} elementi eliminati)")

def _configure_cpu_threads() -> None:
    """
    Su MLP piccoli (64x64) in CPU, 1 thread PyTorch evita l'overhead di
    sincronizzazione OpenMP/futex e accelera forward+backward di 2-4x rispetto a 8-16 thread.
    """
    try:
        torch.set_num_threads(1)
    except Exception:
        pass
    try:
        torch.set_num_interop_threads(1)
    except Exception:
        pass


class LabResult(dict):
    """
    Oggetto risultato restituito da `run_lab`.
    Si comporta come un normale dizionario Python con supporto aggiuntivo all'accesso
    tramite attributi (es. res.run_dir o res["run_dir"]) e include metodi ausiliari
    per la reportistica, il debug e il confronto dei modelli.
    """

    def __getattr__(self, name: str) -> Any:
        try:
            return self[name]
        except KeyError:
            raise AttributeError(f"'LabResult' object has no attribute '{name}'")

    def __setattr__(self, name: str, value: Any) -> None:
        self[name] = value

    def salva_preferito(
        self,
        nome_personalizzato: str | None = None,
        note: str = "",
        tag: str = "Mio Preferito",
    ) -> dict[str, Any]:
        """Salva questo modello nell'archivio dei preferiti personali dell'utente."""
        return salva_nei_preferiti(
            self,
            nome_personalizzato=nome_personalizzato,
            note=note,
            tag=tag,
        )

    def salva_video_preferito(
        self,
        video: int | str | Path = 0,
        nome: str | None = None,
        note: str = "",
    ) -> dict[str, Any]:
        """Salva un video specifico nei preferiti personali dell'utente."""
        target_video = video
        if isinstance(video, int):
            v_list = self.get("success_videos", []) or self.get("recorded_videos", [])
            if v_list and 0 <= video < len(v_list):
                target_video = v_list[video]
            else:
                target_video = self
        return salva_video_preferito(
            video_ref=target_video,
            nome=nome,
            note=note,
        )

    def genera_report(
        self,
        autore: str = "Studente",
        titolo: str = "Report Esperimento Lunar Lander",
        note: str = "",
    ) -> dict[str, Path]:
        """Genera la scheda tecnica riassuntiva in PDF, HTML e Markdown."""
        return genera_report_maturita(
            self,
            autore=autore,
            titolo=titolo,
            note=note,
        )

    def mostra_volo(self, indice: int = 0) -> None:
        """Visualizza uno dei video di prova registrati."""
        videos = self.get("recorded_videos", [])
        if videos and 0 <= indice < len(videos):
            display_notebook_summary(self.get("plot_path"), [videos[indice]])
        else:
            print(f"[play2learn] Nessun video trovato all'indice {indice}.")

    def mostra_evoluzione(self) -> None:
        """Visualizza il timelapse GIF dell'evoluzione dell'agente nel corso del training."""
        p = self.get("gif_evoluzione_path") or (Path(self.get("checkpoint_dir", "")) / "evoluzione_apprendimento.gif")
        if p and Path(p).exists():
            try:
                from IPython.display import Image, display
                print("🎞️ Evoluzione dell'Agente durante il Training (Milestone nel tempo):")
                display(Image(filename=str(p)))
            except Exception:
                print(f"GIF evoluzione salvata in: {p}")
        else:
            print("[play2learn] Nessun timelapse di evoluzione disponibile per questa run.")

    def test_dal_vivo(self, n_episodi: int = 1, seed: int = 999) -> None:
        """Esegue una dimostrazione rapida dal vivo del modello."""
        dimostrazione_live(
            model=self.get("model"),
            env_id=self.get("env_id", ENV_ID),
            reward_fn=self.get("reward_fn"),
            n_episodi=n_episodi,
            seed=seed,
            env_kwargs=self.get("env_kwargs"),
        )


def run_lab(
    reward_fn: Callable,
    mission: str = "luna_standard",
    total_timesteps: int = DEFAULT_TIMESTEPS,
    n_envs: int | None = None,
    seed: int = 42,
    device: str = "cpu",
    env_id: str = ENV_ID,
    gravity: float | None = None,
    enable_wind: bool | None = None,
    wind_power: float | None = None,
    turbulence_power: float | None = None,
    output_dir: str | Path = "runs",
    nome_run: str | None = None,
    n_eval_episodes: int = 3,
    crea_gif: bool = True,
    crea_gif_evoluzione: bool = True,
    display_in_notebook: bool = True,
    use_reward_norm: bool = True,
) -> LabResult:
    """
    Avvia l'addestramento PPO con la funzione di reward personalizzata,
    salva automaticamente checkpoint, telemetria e grafici nella cartella 'runs/',
    e mostra la dashboard completa nel notebook.
    """
    if not _LIBS_AVAILABLE:
        raise ImportError(
            "❌ Librerie mancanti per l'addestramento!\n"
            "Assicurati di aver eseguito la Cella 3 in cima al notebook:\n"
            "!apt-get install -y -qq swig\n"
            "!pip install -q 'gymnasium[box2d]' stable-baselines3 imageio imageio-ffmpeg tqdm rich"
        )

    _configure_cpu_threads()

    # 1. Configurazione fisica della missione
    mission_cfg = MISSIONS.get(mission, MISSIONS["luna_standard"])
    env_kwargs = dict(mission_cfg["env_kwargs"])
    if gravity is not None:
        env_kwargs["gravity"] = float(gravity)
    if enable_wind is not None:
        env_kwargs["enable_wind"] = bool(enable_wind)
    if wind_power is not None:
        env_kwargs["wind_power"] = float(wind_power)
    if turbulence_power is not None:
        env_kwargs["turbulence_power"] = float(turbulence_power)

    # 2. Creazione della directory della run in runs/
    run_dir = create_run_directory(
        mission=mission,
        base_dir=output_dir,
        custom_name=nome_run,
    )
    video_dir = run_dir / "videos"

    # 3. Creazione degli ambienti paralleli ottimizzati + VecNormalize solo reward
    raw_vec_env = make_parallel_envs(
        env_id=env_id,
        reward_fn=reward_fn,
        n_envs=n_envs,
        seed=seed,
        env_kwargs=env_kwargs,
        total_timesteps=total_timesteps,
    )
    actual_n_envs = raw_vec_env.num_envs
    vec_backend = raw_vec_env.__class__.__name__

    vec_env = (
        wrap_vec_normalize(raw_vec_env, gamma=DEFAULT_PPO_PARAMS["gamma"])
        if use_reward_norm
        else raw_vec_env
    )

    print("=" * 76)
    print(f"  🚀 MISSIONE: {mission_cfg['title'].upper()}")
    print(f"  📋 OBIETTIVO: {mission_cfg['description']}")
    print(
        f"  ⚙️  FISICA: Gravità={env_kwargs['gravity']} | Vento={env_kwargs['enable_wind']} "
        f"(Potenza={env_kwargs['wind_power']}, Turbolenza={env_kwargs['turbulence_power']})"
    )
    print(
        f"  ⚡ AMBIENTI: {actual_n_envs} ({vec_backend}) | STEP TOTALI: {total_timesteps:,} | "
        f"RewardNorm: {'ON' if use_reward_norm else 'OFF'}"
    )
    print(f"  📁 SALVATAGGIO RUN IN: runs/{run_dir.name}")
    print("=" * 76)

    # 4. Callback con checkpoint in RAM e progress bar leggera throttled
    tracker = RewardTrackerCallback(
        best_model_save_dir=run_dir,
        total_timesteps=total_timesteps,
        show_progress=True,
        progress_interval_sec=1.5,
        capture_milestones=crea_gif_evoluzione,
    )

    # 5. Creazione modello PPO (verbose=0 per output pulito e veloce)
    model = create_ppo_agent(
        env=vec_env,
        seed=seed,
        device=device,
        verbose=0,
    )

    # 6. Addestramento (progress_bar=False per evitare overhead Rich su Colab)
    print("\n[play2learn] Inizio addestramento dell'agente...")
    t_start = time.perf_counter()
    try:
        model.learn(
            total_timesteps=total_timesteps,
            callback=tracker,
            progress_bar=False,
        )
    except KeyboardInterrupt:
        print("\n[play2learn] Addestramento interrotto manualmente dall'utente.")
    finally:
        vec_env.close()

    elapsed_time = time.perf_counter() - t_start
    completed_steps = max(1, model.num_timesteps)
    fps_avg = int(completed_steps / max(0.1, elapsed_time))
    print(
        f"[play2learn] Addestramento concluso in {elapsed_time:.1f} secondi "
        f"({fps_avg:,} step/sec)!"
    )

    # 7. Salvataggio modello finale + flush su disco del miglior snapshot in RAM
    final_model_path = run_dir / "ppo_lunar_lander.zip"
    model.save(str(final_model_path))

    used_best_weights = tracker.flush_best_model_to_disk(
        keep_best_loaded_if_superior=True,
        superiority_margin=5.0,
    )
    if used_best_weights:
        print(
            f"[play2learn] Ripristinato per la valutazione il miglior checkpoint in corsa "
            f"(picco successo mobile: {tracker.best_moving_success_rate:.1f}%)."
        )

    # 8. Generazione grafico telemetria e diagnosi Flight Coach
    plot_path = run_dir / "reward_plot.png"
    diag_dict = plot_training_results(
        tracker=tracker,
        mission_title=mission_cfg["title"],
        save_path=plot_path,
    )

    # 9. Valutazione imparziale e registrazione video in 2 fasi
    eval_stats = evaluate_and_record(
        model=model,
        env_id=env_id,
        reward_fn=reward_fn,
        video_dir=video_dir,
        target_successes=max(1, n_eval_episodes),
        target_failures=max(1, n_eval_episodes),
        sweep_timeout_seconds=6.0,
        min_eval_episodes=20,
        max_eval_episodes=80,
        seed=seed + 1000,
        env_kwargs=env_kwargs,
        create_gif=crea_gif,
    )

    # 9b. Generazione del timelapse GIF dell'evoluzione dell'apprendimento nel tempo
    gif_evoluzione_path = None
    if crea_gif_evoluzione and tracker.milestone_snapshots:
        try:
            print("\n[play2learn] Generazione del timelapse GIF dell'evoluzione dell'apprendimento...")
            from play2learn_lunarlander.evaluator import crea_gif_evoluzione_apprendimento
            gif_ev_target = run_dir / "evoluzione_apprendimento.gif"
            gif_evoluzione_path = crea_gif_evoluzione_apprendimento(
                model=model,
                milestones=tracker.milestone_snapshots,
                env_id=env_id,
                output_path=gif_ev_target,
                env_kwargs=env_kwargs,
                seed=seed,
            )
            if gif_evoluzione_path and gif_evoluzione_path.exists():
                print(f"🎬 Timelapse evoluzione salvato: runs/{run_dir.name}/{gif_evoluzione_path.name}")
        except Exception as e:
            print(f"[play2learn] Avviso generazione timelapse evoluzione: {e}")

    # 10. Salvataggio automatico metadati e codice sorgente
    meta_path = save_run_metadata(
        run_dir=run_dir,
        mission=mission,
        mission_cfg=mission_cfg,
        env_kwargs=env_kwargs,
        total_timesteps=completed_steps,
        elapsed_seconds=elapsed_time,
        reward_fn=reward_fn,
        eval_stats=eval_stats,
        diagnostics=diag_dict,
    )

    # 11. Stampa riassuntiva dei risultati
    succ_rate = eval_stats.get("eval_success_rate", 0.0)
    n_succ = len(eval_stats.get("success_videos", []))
    n_fail = len(eval_stats.get("failure_videos", []))

    print("\n" + "=" * 76)
    print("  📊 RISULTATI DEI VOLI DI VERIFICA (VALUTAZIONE DETERMINISTICA):")
    print(
        f"  • Tasso di Allunaggio Riuscito: {succ_rate:.1f}% "
        f"({eval_stats.get('eval_successes', 0)} su {eval_stats.get('total_evaluated_episodes', 0)} prove)"
    )
    print(f"  • Video registrati:            {n_succ} allunaggi riusciti 🟢 | {n_fail} fallimenti 🔴")
    print(
        f"  • Durata Media Volo:           {eval_stats['mean_length']:.0f} step "
        f"({eval_stats['mean_length'] / 50:.1f} s)"
    )
    print(
        f"  • Carburante Medio Speso:      {eval_stats['mean_fuel']:.1f} unità "
        f"(Riusciti: {eval_stats.get('mean_fuel_success', 0):.1f} un.)"
    )
    print(
        f"  • Custom Reward Media:         {eval_stats['mean_reward']:.1f} ± "
        f"{eval_stats['std_reward']:.1f}"
    )
    print(f"  💾 Tutti i checkpoint e i video sono archiviati in: runs/{run_dir.name}")
    print("=" * 76 + "\n")

    # 12. GESTIONE OBIETTIVO RAGGIUNTO (SUCCESSO >= 80%): Archiviazione in 'runs/successi/'
    report_bundle = None
    if succ_rate >= 80.0:
        print("🎉 " + "=" * 72)
        print("  🏆 OBIETTIVO ALLUNAGGIO RAGGIUNTO! (Tasso di successo >= 80%)")
        print("  Archiviazione automatica in 'runs/successi/' e generazione Scheda Tecnica...")
        print("=" * 76)

        succ_info = archivia_esperimento_successo(
            run_dir=run_dir,
            nome_run=nome_run or mission,
            succ_rate=succ_rate,
            eval_stats=eval_stats,
        )
        succ_dir = succ_info["dest_dir"]
        rep_files = succ_info["rep_files"]
        report_bundle = {
            "succ_entry": succ_info["entry"],
            "rep_files": rep_files,
            "dir_path": succ_dir,
        }

        _display_success_popup(succ_info["entry"], rep_files)

    # 13. Mostra dashboard e video nel Notebook
    if display_in_notebook:
        display_notebook_summary(
            plot_path=plot_path,
            videos=eval_stats.get("videos"),
            success_videos=eval_stats.get("success_videos"),
            failure_videos=eval_stats.get("failure_videos"),
            gif_path=gif_evoluzione_path or eval_stats.get("gif_path"),
        )

    return LabResult({
        "run_id": run_dir.name,
        "run_dir": run_dir,
        "checkpoint_dir": run_dir,
        "model_path": final_model_path,
        "best_model_path": tracker.best_model_path or final_model_path,
        "plot_path": plot_path,
        "recorded_videos": eval_stats.get("videos", []),
        "success_videos": eval_stats.get("success_videos", []),
        "failure_videos": eval_stats.get("failure_videos", []),
        "gif_path": eval_stats.get("gif_path"),
        "gif_evoluzione_path": gif_evoluzione_path,
        "eval_success_rate": eval_stats.get("eval_success_rate", 0.0),
        "eval_mean_length": eval_stats.get("mean_length", 0.0),
        "eval_mean_fuel": eval_stats.get("mean_fuel", 0.0),
        "eval_stats": eval_stats,
        "diagnosis": diag_dict.get("formatted_text", ""),
        "elapsed_seconds": elapsed_time,
        "total_timesteps": total_timesteps,
        "metadata_path": meta_path,
        "model": model,
        "reward_fn": reward_fn,
        "env_id": env_id,
        "env_kwargs": env_kwargs,
        "report_bundle": report_bundle,
        "maturita_bundle": report_bundle,
    })

def dimostrazione_live(
    model: PPO,
    env_id: str = ENV_ID,
    reward_fn: Callable | None = None,
    n_episodi: int = 1,
    seed: int = 999,
    env_kwargs: dict[str, Any] | None = None,
) -> None:
    """
    Esegue uno o più episodi dimostrativi stampando a video la telemetria
    in tempo reale per osservare il comportamento dinamico dell'agente.
    """
    if not _LIBS_AVAILABLE:
        raise ImportError(
            "❌ Librerie mancanti per la dimostrazione!\n"
            "Assicurati di aver eseguito la Cella 3 in cima al notebook:\n"
            "!apt-get install -y -qq swig\n"
            "!pip install -q 'gymnasium[box2d]' stable-baselines3 imageio imageio-ffmpeg tqdm rich"
        )

    env_kwargs = env_kwargs or {}
    env = gym.make(env_id, **env_kwargs)
    if reward_fn is not None:
        env = CustomRewardWrapper(env, reward_fn)

    print("\n" + "=" * 65)
    print("  🎬 DIMOSTRAZIONE DI ALLUNAGGIO IN DIRETTA")
    print("=" * 65)

    for ep in range(n_episodi):
        obs, _ = env.reset(seed=seed + ep)
        done = False
        step = 0
        tot_fuel = 0.0

        while not done:
            action, _ = model.predict(obs, deterministic=True)
            obs, rew, term, trunc, info = env.step(action)
            step += 1
            done = term or trunc

            act_int = int(action)
            if act_int == 2:
                tot_fuel += 1.0
            elif act_int in (1, 3):
                tot_fuel += 0.3

            if step % 35 == 0 or done:
                x, y, vx, vy, angle = obs[0], obs[1], obs[2], obs[3], obs[4]
                print(
                    f"  Step {step:3d} | x={x:+.2f} | y={y:.2f} | vy={vy:+.2f} | "
                    f"angolo={angle:+.2f} rad | fuel={tot_fuel:.1f}"
                )

        is_succ = info.get("is_success", False) if isinstance(info, dict) else False
        esito_str = "🎯 ALLUNAGGIO RIUSCITO CON SUCCESSO!" if is_succ else "💥 SCHIANTO / IMPATTO"
        print(f"\n  ESITO VOLO #{ep + 1}: {esito_str}")
        print(f"  Durata: {step} step ({step / 50:.1f} s) | Carburante totale speso: {tot_fuel:.1f} unità")
        print("-" * 65)

    env.close()
